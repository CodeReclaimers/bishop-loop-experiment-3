        raise JSONParseError(f"expected '\"' at {pos}")

    n = len(text)
    # Precompile the regex for string escape handling
    _ESCAPE_PATTERN = re.compile(r'''
        \\(u[0-9a-fA-F]{4}|[0-367bfnrtv"\\/])
    ''', re.VERBOSE)

    def _unescape(match):
        esc = match.group(1)
        if esc.startswith('u'):
            cp = int(esc[1:], 16)
            if 0xD800 <= cp <= 0xDBFF:
                # Handle surrogate pair
                next_match = _ESCAPE_PATTERN.search(text, match.end())
                if next_match and next_match.group(1).startswith('u'):
                    cp2 = int(next_match.group(1)[1:], 16)
                    if 0xDC00 <= cp2 <= 0xDFFF:
                        cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                        return chr(cp)
            return chr(cp)
        return _ESCAPE_CHARS[esc]

    # Find the closing quote
    end_quote = text.find('"', pos)
    if end_quote == -1:
        raise JSONParseError("unterminated string")
    
    # Extract the string content and process escapes
    content = text[pos:end_quote]
    result = _ESCAPE_PATTERN.sub(_unescape, content)
