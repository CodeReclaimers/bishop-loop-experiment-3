        raise JSONParseError(f"expected '\"' at {pos}")
    pos = pos + 1
    n = len(text)
    # Find the closing quote
    end_pos = text.find('"', pos)
    if end_pos == -1:
        raise JSONParseError("unterminated string")
    # Extract the raw content between quotes
    raw_content = text[pos:end_pos]
    
    def replace_escape(match):
        esc = match.group(0)[1]  # Get the character after backslash

                # The regex is not designed to handle this, so we need to
                # manually check for the next surrogate
                return match.group(0)  # Return original to handle manually
            return chr(cp)
        else:
            raise JSONParseError(f"bad escape \\{esc} at {match.start()}")
