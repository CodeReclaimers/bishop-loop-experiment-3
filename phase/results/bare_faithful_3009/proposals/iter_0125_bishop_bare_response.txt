# Precomputed whitespace jump array
def _build_ws_jump_array(text: str) -> list[int]:
    """Build a jump array for skipping whitespace in constant time."""
    n = len(text)
    jump = [0] * n
    i = n - 1
    while i >= 0:
        if text[i] in _WS_CHARS:
            if i + 1 < n:
                jump[i] = jump[i + 1] + 1
            else:
                jump[i] = 1
        else:
            jump[i] = 0
        i -= 1
    return jump


def _skip_ws_jump(text: str, pos: int, jump: list[int]) -> int:
    """Skip whitespace using precomputed jump array."""
    if pos >= len(text):
        return pos
    return pos + jump[pos]

def _parse_object(text: str, pos: int) -> tuple[dict, int]:
    if text[pos] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos = pos + 1
    out: dict = {}
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "}":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != '"':
            raise JSONParseError(f"expected string key at {pos}")
        # Use string.find to locate closing quote
        key_start = pos + 1
        pos = key_start
        while True:
            quote_pos = text.find('"', pos)
