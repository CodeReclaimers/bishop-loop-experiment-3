def _parse_object(text: str, pos: int) -> tuple[dict, int]:
    if text[pos] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos = pos + 1
    out: dict = {}
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "}":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != '"':
            raise JSONParseError(f"expected string key at {pos}")
        # Use string.find to locate closing quote
        key_start = pos + 1
        pos = key_start
        while True:
            quote_pos = text.find('"', pos)
            if quote_pos == -1:
                raise JSONParseError("unterminated string")
            # Check if the quote is escaped
            if text[quote_pos - 1] != "\\":
                key = text[key_start:quote_pos]
                pos = quote_pos + 1
                break
            pos = quote_pos + 1
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != ":":
            raise JSONParseError(f"expected ':' at {pos}")

        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)

        if c == ",":
            pos = pos + 1
