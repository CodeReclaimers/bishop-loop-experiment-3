def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    if text[pos] == "0":
        pos += 1
    elif text[pos] in _DIGIT_CHARS:
        # State machine for number parsing
        state = 0  # 0: integer part, 1: decimal point, 2: fractional part, 3: exponent
        while pos < n:
            c = text[pos]
            if state == 0 and c in _DIGIT_CHARS:
                pos += 1
            elif state == 0 and c == ".":
                state = 1
                pos += 1
            elif state == 1 and c in _DIGIT_CHARS:
                state = 2
                pos += 1
            elif state == 2 and c in _DIGIT_CHARS:
                pos += 1
            elif state == 0 and c in ("e", "E"):
                state = 3
                pos += 1
                if pos < n and text[pos] in ("+", "-"):
                    pos += 1
            elif state == 3 and c in _DIGIT_CHARS:
                pos += 1
            else:
                break
    else:
        raise JSONParseError(f"bad number at {pos}")
    s = text[start:pos]
    # Check if it's a float by looking for decimal point or exponent
    is_float = "." in s or "e" in s.lower()
    if is_float:
        return float(s), pos
    return int(s), pos
