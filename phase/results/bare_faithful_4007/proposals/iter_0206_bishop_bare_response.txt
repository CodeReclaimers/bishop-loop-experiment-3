def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    # Handle optional minus sign
    if text[pos] == '-':
        pos += 1
        if pos >= n:
            raise JSONParseError(f"bad number at {start}")
    
    # Parse integer part
    if text[pos] == '0':
        pos += 1
    else:
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
        if pos == start + 1 and text[start] == '0':
            raise JSONParseError(f"bad number at {start}")
    
    # Parse fractional part
    if pos < n and text[pos] == '.':
        pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number at {start}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    
    # Parse exponent
    if pos < n and text[pos] in 'eE':
        pos += 1
        if pos < n and text[pos] in '+-':
            pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number at {start}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    
    s = text[start:pos]
    if '.' in s or 'e' in s.lower():
        return float(s), pos
    return int(s), pos
