def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    # Skip sign if present
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:

    # Scan integer part

        pos += 1
    # Check for fractional part
    if pos < n and text[pos] == ".":
