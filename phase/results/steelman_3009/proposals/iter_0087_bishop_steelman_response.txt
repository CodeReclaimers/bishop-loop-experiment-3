def _skip_ws(text: str, pos: int) -> int:
    # Use str.translate to efficiently skip leading whitespace
    # by removing all whitespace chars from the start of the string
    # and calculating the new position from the difference
    if pos >= len(text):
        return pos
    # Create translation table to delete whitespace chars
    whitespace_trans = str.maketrans('', '', ''.join(_WS_CHARS))
    # Strip leading whitespace from the substring starting at pos
    stripped = text[pos:].translate(whitespace_trans)
    # Calculate new position as original pos plus the difference in length
    new_pos = pos + (len(text[pos:]) - len(stripped))
