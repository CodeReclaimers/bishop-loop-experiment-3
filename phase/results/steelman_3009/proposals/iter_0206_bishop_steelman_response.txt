This would require repeatedly calling `str.find` for each whitespace character, negating the performance benefit. Additionally, it doesn't handle the case where the entire string consists of whitespace, which would cause `str.find` to return -1 and break the logic.

def _skip_ws(text: str, pos: int) -> int:
    # Use str.translate to efficiently skip whitespace by replacing
    # all whitespace characters with a sentinel character, then find
    # the first non-sentinel character
    if pos >= len(text):
        return pos
    # Create translation table to replace whitespace with a sentinel
    trans_table = str.maketrans({ch: '\x00' for ch in _WS_CHARS})
    # Translate the substring starting at pos
    translated = text[pos:].translate(trans_table)
    # Find first non-sentinel character
    new_pos = translated.find('\x00')
    if new_pos == -1:
        # All characters are whitespace
        return len(text)
    return pos + new_pos
