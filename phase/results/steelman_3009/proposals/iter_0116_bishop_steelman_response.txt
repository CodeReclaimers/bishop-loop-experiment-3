def _skip_ws(text: str, pos: int) -> int:
    # Use a set-based lookup for whitespace characters for O(1) membership
    # Precompute the bytes once to avoid repeated encoding overhead
    if pos >= len(text):
        return pos
    # Convert to bytes for fast lookup, but only for the remaining string
    remaining_bytes = text[pos:].encode('utf-8')
    # Find the first non-whitespace byte
    byte_pos = 0
    while byte_pos < len(remaining_bytes):
        byte_val = remaining_bytes[byte_pos]
        # Check if this byte corresponds to a whitespace character
        if byte_val in (0x20, 0x09, 0x0A, 0x0D):  # space, tab, newline, carriage return
            byte_pos += 1
