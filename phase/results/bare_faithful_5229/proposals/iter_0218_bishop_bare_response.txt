def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    
    # State machine for number parsing
    state = 0  # 0: integer part, 1: decimal point, 2: fractional part, 3: exponent
    while pos < n:
        c = text[pos]
        if state == 0:  # integer part
            if c == "0":
                pos += 1
            elif c in _DIGIT_CHARS:
                while pos < n and text[pos] in _DIGIT_CHARS:
                    pos += 1
                state = 1
            else:
                raise JSONParseError(f"bad number at {pos}")
        elif state == 1:  # decimal point
            if c == ".":
                state = 2
                pos += 1
            elif c in ("e", "E"):
                state = 3
                pos += 1
            else:
                break
        elif state == 2:  # fractional part
            if c in _DIGIT_CHARS:
                while pos < n and text[pos] in _DIGIT_CHARS:
                    pos += 1
                state = 3
            else:
                raise JSONParseError(f"bad number fractional at {pos}")
        elif state == 3:  # exponent
            if c in ("+", "-"):
                pos += 1
            while pos < n and text[pos] in _DIGIT_CHARS:
                pos += 1
            break
        else:
            break
    
    # Handle case where we're at the end of the integer part
    if state == 0:
        # We only saw an integer part, so we're done
        pass
    elif state == 1 and pos < n and text[pos] in ("e", "E"):
        # We have an integer part and an exponent
        state = 3
        pos += 1
        if pos < n and text[pos] in ("+", "-"):
            pos += 1
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    
    s = text[start:pos]
    is_float = ("." in s) or ("e" in s.lower())
    if is_float:
        return float(s), pos
    return int(s), pos
