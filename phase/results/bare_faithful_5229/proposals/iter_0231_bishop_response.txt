We can achieve this by using a `list` instead of concatenating strings directly in each iteration of the loop. Instead, we can collect the characters in a list and then convert it to a string only when needed at the end.

Here's how you can modify the `_parse_string` function:

def _parse_string(text: str, pos: int) -> tuple[str, int]:
    out = []
    esc = None
    while True:
        c = text[pos]
        if c == '"':
            return "".join(out), pos + 1
        elif c == "\\":
            esc = text[pos + 1]
            pos += 2
        else:
            if esc is not None:
                match esc:
                    case "n":
                        out.append("\n")
                    case "r":
                        out.append("\r")
                    case "t":
                        out.append("\t")
                    case "u":
                        if pos + 4 >= len(text):
                            raise JSONParseError(f"bad unicode escape at {pos}")
                        hex4 = text[pos + 1 : pos + 5]
                        if not _is_hex4(hex4):
                            raise JSONParseError(f"bad unicode escape at {pos}")
                        cp = int(hex4, 16)
                        if 0xD800 <= cp <= 0xDBFF:
                            if pos + 5 >= len(text) or text[pos + 5] != "\\" or pos + 6 >= len(text) or text[pos + 6] != "u":
                                raise JSONParseError(f"unpaired high surrogate at {pos}")
                            if pos + 10 >= len(text):
                                raise JSONParseError(f"bad low surrogate at {pos}")
                            hex4b = text[pos + 7 : pos + 11]
                            if not _is_hex4(hex4b):
                                raise JSONParseError(f"bad low surrogate at {pos + 7}")
                            cp2 = int(hex4b, 16)
                            if not (0xDC00 <= cp2 <= 0xDFFF):
                                raise JSONParseError(f"bad low surrogate at {pos + 7}")
                            cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                            out.append(chr(cp))
                            pos += 11
                            continue
                        out.append(chr(cp))
                        pos = pos + 4
                    else:
                        raise JSONParseError(f"bad escape \\{esc} at {pos}")
                esc = None
            elif c < '\x20':
                raise JSONParseError(f"unescaped control character at {pos}")
            out.append(c)
            pos += 1
    raise JSONParseError("unterminated string")

In this modified version, we use a list `out` to collect the characters. When we encounter a closing double quote `"`, we convert the list `out` to a string using `"".join(out)` and return it along with the position `pos + 1`. This approach avoids the overhead of concatenating strings directly in each iteration, resulting in faster parsing times.
