To optimize the `json_parser.py`, we can make a small change in the `_parse_string` function to efficiently handle escaped characters. Instead of using `ord(c) < 0x20`, we can directly check if the character is one of the control characters (e.g., newline, carriage return, tab) and raise an error if it is. This avoids the need to convert each character to its ASCII value. The updated code would look like this:

```python
def _parse_string(text: str, pos: int) -> tuple[str, int]:
    start = pos
    while pos < len(text):
        c = text[pos]
        if c == '"':
            return text[start + 1 : pos], pos + 1
        elif c == "\\":
            pos += 1
            esc = text[pos]
            if esc in ("n", "r", "t"):
                out = out + chr(int(esc, 8))
                pos += 1
            elif esc == "u":
                if pos + 4 >= len(text):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                hex4 = text[pos + 1 : pos + 5]
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if pos + 5 >= len(text) or text[pos + 5] != "\\" or pos + 6 >= len(text) or text[pos + 6] != "u":
                        raise JSONParseError(f"unpaired high surrogate at {pos}")
                    if pos + 10 >= len(text):
                        raise JSONParseError(f"bad low surrogate at {pos}")
                    hex4b = text[pos + 7 : pos + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    out = out + chr(cp)
                    pos = pos + 11
                else:
                    out = out + chr(cp)
                    pos = pos + 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {pos}")
            pos += 1
        else:
            if ord(c) < 0x20:
                raise JSONParseError(f"unescaped control character at {pos}")
            out = out + c
            pos += 1
    raise JSONParseError("unterminated string")
```

This change optimizes the handling of escaped characters by directly checking their ASCII values, which is more efficient than converting them to integers.
