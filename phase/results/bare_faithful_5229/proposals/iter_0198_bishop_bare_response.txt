    stack = [(out, False)]  # (current_dict, expecting_comma)
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "}":
        return out, pos + 1
    while stack:
        current_dict, expecting_comma = stack[-1]
        if expecting_comma:
            pos = _skip_ws(text, pos)
            if pos >= len(text):
