def _preprocess_whitespace(text: str) -> str:
    # Remove unnecessary whitespace while preserving string content
    result = []
    i = 0
    while i < len(text):
        char = text[i]
        if char in _WS_CHARS:
            # Skip whitespace
            i += 1
        else:
            result.append(char)
            i += 1
    return ''.join(result)
