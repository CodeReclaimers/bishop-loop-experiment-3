# Precompute a skip table for common whitespace patterns
def _build_skip_table(text: str) -> list[int]:
    table = [0] * len(text)
    i = len(text) - 1
    while i >= 0:
        if text[i] in ' \t\n\r':
            table[i] = 1 + (table[i + 1] if i + 1 < len(text) else 0)
        else:
            table[i] = 0
        i -= 1
    return table
