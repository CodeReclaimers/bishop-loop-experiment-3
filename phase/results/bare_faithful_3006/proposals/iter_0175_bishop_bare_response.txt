def _parse_value(text: str, pos: int) -> tuple[Any, int]:
    while pos < len(text) and text[pos] in _WS_CHARS:
        pos += 1
    if pos >= len(text):
        raise JSONParseError("unexpected end of input")
    c = text[pos]
    parser_map = {
        "{": _parse_object,
        "[": _parse_array,
        '"': _parse_string,
        "t": _parse_bool,
        "f": _parse_bool,
        "n": _parse_null,
        "-": _parse_number,
    }
    for digit in _DIGIT_CHARS:
        parser_map[digit] = _parse_number
    parser = parser_map.get(c)
    if parser is None:
        raise JSONParseError(f"unexpected character {c!r} at position {pos}")
    return parser(text, pos)
