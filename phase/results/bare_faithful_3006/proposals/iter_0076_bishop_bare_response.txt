def _skip_ws(text: str, pos: int) -> int:
    # Use a sequence of three calls to text.find for each whitespace character,
    # returning the smallest index among them
    while pos < len(text):
        found = text.find('\n', pos)
        if found == -1:
            found = text.find('\r', pos)
            if found == -1:
                found = text.find(' ', pos)
                if found == -1:
                    found = text.find('\t', pos)
                    if found == -1:
                        break
        if found == pos:
            pos += 1
        else:
            break
    return pos
