def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    # Skip leading whitespace
    while pos < n and text[pos] in _WS_CHARS:
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    # Find end of number
    while pos < n and (text[pos] in _DIGIT_CHARS or text[pos] in ".eE+-"):
        pos += 1
    s = text[start:pos]
    try:
        if '.' in s or 'e' in s.lower():
            return float(s), pos
        else:
            return int(s), pos
    except ValueError:
        raise JSONParseError(f"bad number at {start}")
