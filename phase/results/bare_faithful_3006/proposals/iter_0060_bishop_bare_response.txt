def _skip_ws(text: str, pos: int) -> int:
    # Replace character-by-character scan with single pass using str.find
    if not _WS_CHARS:
        return pos
    min_pos = len(text)
    for ws_char in _WS_CHARS:
        found_pos = text.find(ws_char, pos)
        if found_pos != -1 and found_pos < min_pos:
            min_pos = found_pos
    if min_pos == len(text):
        return pos
    # Skip all whitespace characters at the found position
