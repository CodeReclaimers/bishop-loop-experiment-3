def _skip_ws(text: str, pos: int) -> int:
    # Use text.find to jump directly to the next non-whitespace character
    while pos < len(text):
        next_space = text.find(' ', pos)
        next_tab = text.find('\t', pos)
        next_newline = text.find('\n', pos)
        next_cr = text.find('\r', pos)
        
        # Find the minimum position of any whitespace character
        min_pos = len(text)
        if next_space != -1:
            min_pos = min(min_pos, next_space)
        if next_tab != -1:
            min_pos = min(min_pos, next_tab)
        if next_newline != -1:
            min_pos = min(min_pos, next_newline)
        if next_cr != -1:
            min_pos = min(min_pos, next_cr)
        
        # If no whitespace found, we're done
        if min_pos == len(text):
            return pos
        
        # If we found whitespace at current position, continue
        if min_pos == pos:
            pos = min_pos + 1
        else:
            # Found non-whitespace at min_pos, return it
            return min_pos
