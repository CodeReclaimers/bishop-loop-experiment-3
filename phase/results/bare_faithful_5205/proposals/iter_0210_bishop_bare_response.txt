def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    
    # State machine for number parsing
    state = "integer"  # "integer", "decimal", "exponent"
    while pos < n:
        c = text[pos]
        if state == "integer":
            if c == "0":
                pos += 1
            elif c in _DIGIT_CHARS:
                pos += 1
            elif c == ".":
                state = "decimal"
                pos += 1
            elif c in ("e", "E"):
                state = "exponent"
                pos += 1
            else:
                break
        elif state == "decimal":
            if c in _DIGIT_CHARS:
                pos += 1
            elif c in ("e", "E"):
                state = "exponent"
                pos += 1
            else:
                break
        elif state == "exponent":
            if c in ("+", "-"):
                pos += 1
            elif c in _DIGIT_CHARS:
                pos += 1
            else:
                break
        else:
            break
    
    # Handle the case where we're at the end of the number
    if pos < n and text[pos] in ("e", "E"):
        state = "exponent"
        pos += 1
        if pos < n and text[pos] in ("+", "-"):
            pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number exponent at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    
    s = text[start:pos]
    is_float = state == "decimal" or state == "exponent"
    if is_float:
        return float(s), pos
    return int(s), pos
