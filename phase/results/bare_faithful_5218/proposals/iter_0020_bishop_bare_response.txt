def _parse_value(text: str, pos: int) -> tuple[Any, int]:
    pos = _skip_ws(text, pos)
    if pos >= len(text):
        raise JSONParseError("unexpected end of input")
    c = text[pos]
    # State machine approach: use a dictionary to map first characters to parsing functions
    parser_map = {
        '{': _parse_object,
        '[': _parse_array,
        '"': _parse_string,
        't': _parse_bool,
        'f': _parse_bool,
        'n': _parse_null,
        '-': _parse_number,
    }
    # Add digit characters to the parser map
    for digit in _DIGIT_CHARS:
        parser_map[digit] = _parse_number
    
    if c in parser_map:
        return parser_map[c](text, pos)
