```python
def _parse_string(text: str, pos: int) -> tuple[str, int]:
    n = len(text)
    start = pos + 1
    end = start
    while end < n and text[end] not in _WS_CHARS and text[end] != '"':
        end += 1
    if end == n:
        raise JSONParseError(f"unterminated string at {pos}")
    return text[start:end], end + 1
```
