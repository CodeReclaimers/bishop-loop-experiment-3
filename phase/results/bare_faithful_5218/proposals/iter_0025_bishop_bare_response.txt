def _skip_ws(text: str, pos: int) -> int:
    # DFA-based whitespace skipping using a lookup table
    n = len(text)
    # Precomputed DFA transition table for whitespace characters
    # 0 = not whitespace, 1 = whitespace
    ws_dfa = [0] * 256
    for i in range(256):
        if chr(i) in _WS_CHARS:
            ws_dfa[i] = 1
    while pos < n and ws_dfa[ord(text[pos])] == 1:
        pos += 1
    return pos
