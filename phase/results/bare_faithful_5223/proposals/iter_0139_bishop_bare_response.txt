def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    # Single loop to scan all number characters
    while pos < n:
        c = text[pos]
        if c in _DIGIT_CHARS or c == "." or c in "eE" or c in "+-":

                is_float = True
            elif c in "eE":
