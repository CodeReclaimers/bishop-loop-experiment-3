class ParsedJSON:
    """A wrapper around parsed JSON data that supports lazy parsing."""
    def __init__(self, data: Any):
        self._data = data

    @property
    def data(self) -> Any:
        return self._data


def parse_partial(text: str, pos: int = 0) -> tuple[ParsedJSON, int]:
