def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    # Dynamic programming approach for number parsing
    num = 0
    if text[pos] == "0":
        pos += 1
    elif text[pos] in _DIGIT_CHARS:
        while pos < n and text[pos] in _DIGIT_CHARS:
            num = num * 10 + (ord(text[pos]) - ord('0'))
            pos += 1
    else:
        raise JSONParseError(f"bad number at {pos}")
    is_float = False
    if pos < n and text[pos] == ".":
        is_float = True
        pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number fractional at {pos}")
        # Dynamic programming approach for fractional part
        frac = 0
        frac_digits = 0
        while pos < n and text[pos] in _DIGIT_CHARS:
            frac = frac * 10 + (ord(text[pos]) - ord('0'))
            frac_digits += 1
            pos += 1
        # Convert fractional part to float
        num = num + frac / (10 ** frac_digits)
    if pos < n and text[pos] in ("e", "E"):
        is_float = True
        pos += 1
        if pos < n and text[pos] in ("+", "-"):
            pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number exponent at {pos}")
        # Dynamic programming approach for exponent part
        exp = 0
        while pos < n and text[pos] in _DIGIT_CHARS:
            exp = exp * 10 + (ord(text[pos]) - ord('0'))
            pos += 1
        # Apply exponent
        num = num * (10 ** exp)
    if is_float:
        return float(num), pos
    return int(num), pos
