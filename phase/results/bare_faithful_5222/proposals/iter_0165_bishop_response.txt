def _skip_ws(text: str, pos: int) -> int:
    n = len(text)
    # Use str.lstrip() for efficient whitespace skipping
    pos = pos + len(text[pos:n].lstrip(' \t\n\r'))
    return pos


def _parse_value(text: str, pos: int) -> tuple[Any, int]:
    pos = _skip_ws(text, pos)
    if pos >= len(text):
        raise JSONParseError("unexpected end of input")
    c = text[pos]
    if c == "{":
        return _parse_object(text, pos)
    if c == "[":
        return _parse_array(text, pos)
    if c == '"':
        yield from _parse_string(text, pos)

This change uses a generator expression `yield from _parse_string(text, pos)` to directly yield the parsed string value without building it in memory. This approach reduces the memory footprint and improves performance when parsing large JSON documents.
