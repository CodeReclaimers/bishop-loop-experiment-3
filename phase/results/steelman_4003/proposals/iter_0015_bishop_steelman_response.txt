    start = pos
    # Handle optional minus sign
    if text[pos] == "-":
        pos += 1
        if pos >= len(text):
            raise JSONParseError(f"bad number at {start}")
    
    # Parse integer part
    if text[pos] == "0":
        pos += 1
    else:
        while pos < len(text) and text[pos] in _DIGIT_CHARS:
            pos += 1
        if pos == start + 1 and text[start] == "0":
