import re

# Precompile the number regex for performance
_NUMBER_RE = re.compile(r"""
    -?          # optional minus
    (?:0|[1-9]\d*)  # integer part: zero or non-zero number
    (?:\.\d+)?  # optional fractional part
    (?:[eE][+-]?\d+)?  # optional exponent
""", re.VERBOSE)

def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    # Find the longest match starting at pos
    match = _NUMBER_RE.match(text, pos)
    if not match:
        raise JSONParseError(f"bad number at {pos}")
    s = match.group(0)
    end_pos = pos + len(s)
    try:
        # Check if it's a float by presence of decimal point or exponent
        if '.' in s or 'e' in s.lower():
            return float(s), end_pos
        return int(s), end_pos
    except ValueError:
