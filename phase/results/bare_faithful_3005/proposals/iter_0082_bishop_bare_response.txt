    pos = _skip_ws(text, 0, ws_indices)
    if pos >= len(text):

def _compute_whitespace_indices(text: str) -> list[int]:
    """Pre-compute indices of all whitespace characters."""
    indices = []
    for i, c in enumerate(text):
        if c in ' \t\n\r':
            indices.append(i)
    return indices


def _skip_ws(text: str, pos: int, ws_indices: list[int]) -> int:
    """Skip whitespace using pre-computed indices."""
    # Binary search for the first whitespace character >= pos
    left, right = 0, len(ws_indices) - 1
    result = pos
    while left <= right:
        mid = (left + right) // 2
        idx = ws_indices[mid]
        if idx >= pos:
            result = idx + 1
            right = mid - 1
        else:
            left = mid + 1
    return result
