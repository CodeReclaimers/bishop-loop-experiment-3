def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos = pos + 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    # Use built-in conversion with exception handling to find end of literal
    try:
        # Try to parse as int first
        num = int(text[start:pos])
        # If we successfully parsed an int, check if it's actually a float
        is_float = False
        if pos < n and text[pos] == ".":
            is_float = True
        elif pos < n and text[pos] in ("e", "E"):
            is_float = True
        if not is_float:
            return num, pos
    except ValueError:
        # If int parsing fails, it's definitely a float
        is_float = True
    
    # Find end of numeric literal using built-in conversion
    while pos < n:
        try:
            # Try to parse from start to current position
            s = text[start:pos]
            if is_float:
                float(s)
            else:
                int(s)
            pos = pos + 1
        except ValueError:
            # We've gone too far, so the previous position was the end
            pos = pos - 1
            break
    
    # Parse the final number
    s = text[start:pos]
    if is_float:
        return float(s), pos
    return int(s), pos
