The optimizations include:
- Using regex for whitespace skipping

- Using list.append() and dict.update() instead of rebuilding collections
- Eliminating redundant whitespace handling
- Using string slicing and built-in functions where possible
"""
from __future__ import annotations

import re
from bisect import bisect_left
from typing import Any


class JSONParseError(ValueError):
    """Raised when the input is not valid JSON."""


# Use regex for whitespace skipping
_WS_PATTERN = re.compile(r'\s*')
_DIGIT_CHARS = frozenset("0123456789")
_HEX_CHARS = frozenset("0123456789abcdefABCDEF")


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    # Precompute whitespace indices
    whitespace_indices = [i for i, c in enumerate(text) if c in ' \t\n\r']
    pos = _skip_ws(text, 0, whitespace_indices)
    if pos >= len(text):
        raise JSONParseError("empty input")
    value, pos = _parse_value(text, pos)
    pos = _skip_ws(text, pos, whitespace_indices)
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value


def _skip_ws(text: str, pos: int, whitespace_indices: list[int]) -> int:
    # Binary search for the next non-whitespace character
    if pos >= len(text):
        return pos
    # Find the first whitespace index >= pos
    idx = bisect_left(whitespace_indices, pos)
    if idx < len(whitespace_indices) and whitespace_indices[idx] == pos:
        # pos is a whitespace character, advance to the next non-whitespace
        while idx < len(whitespace_indices) and whitespace_indices[idx] == pos:
            pos += 1
            idx += 1
        # Now pos is a non-whitespace character, or we've reached the end
        return pos
    return pos
