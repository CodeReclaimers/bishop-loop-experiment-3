    # Pre-compute whitespace indices once in parse

    # Binary search for the first non-whitespace character
    low, high = 0, len(_skip_ws.whitespace_indices) - 1
    while low <= high:
        mid = (low + high) // 2
        if _skip_ws.whitespace_indices[mid] >= pos:
