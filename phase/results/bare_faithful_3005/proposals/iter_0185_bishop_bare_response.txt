def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos = pos + 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    # Find the end of the number substring
    end = pos
    while end < n and text[end] in _DIGIT_CHARS:
        end += 1
    # Check for fractional part
    if end < n and text[end] == ".":
        end += 1
        while end < n and text[end] in _DIGIT_CHARS:
            end += 1
    # Check for exponent
    if end < n and text[end] in ("e", "E"):
        end += 1
        if end < n and text[end] in ("+", "-"):
            end += 1
        while end < n and text[end] in _DIGIT_CHARS:
            end += 1
    s = text[start:end]
    if "." in s or "e" in s.lower():

    return int(s), end
