def _skip_ws(text: str, pos: int) -> int:
    # Binary search lookup into pre-computed whitespace indices
    if pos >= len(text):
        return pos
    # Check if we have a cached list of whitespace positions for this text
    if not hasattr(_skip_ws, 'whitespace_indices'):
        _skip_ws.whitespace_indices = []
        for i, char in enumerate(text):
            if char in ' \t\n\r':
                _skip_ws.whitespace_indices.append(i)
        _skip_ws.whitespace_indices.append(len(text))  # Sentinel
    
    # Binary search for the first whitespace index >= pos
    indices = _skip_ws.whitespace_indices
    left, right = 0, len(indices) - 1
    while left <= right:
        mid = (left + right) // 2
        if indices[mid] < pos:
            left = mid + 1
        else:
            right = mid - 1
    
    # If we found a whitespace index at or after pos, skip to the next non-whitespace
    if left < len(indices) and indices[left] == pos:
        # Skip all consecutive whitespace
        while left < len(indices) and indices[left] == pos:
            pos = indices[left] + 1
            left += 1
        # Return the position after the last whitespace character
        if left < len(indices):
            return indices[left]
        else:
            return len(text)
    else:
        # pos is not a whitespace character, so return it as is
        return pos
