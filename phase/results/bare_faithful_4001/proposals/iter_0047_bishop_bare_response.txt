def _skip_ws(text: str, pos: int) -> int:
    # Sliding window technique for whitespace skipping
    window_size = 8
    while pos < len(text):
        # Check window of characters
        window_end = min(pos + window_size, len(text))
        window = text[pos:window_end]
        for i, c in enumerate(window):
            if c not in _WS_CHARS:
                return pos + i
        pos = window_end
    return pos
