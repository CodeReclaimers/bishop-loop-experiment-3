def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    # State machine for number parsing
    state = 0  # 0: start, 1: int part, 2: frac part, 3: exp part
    while pos < n:
        c = text[pos]
        if state == 0:
            if c == "-":
                pos += 1
            elif c == "0":
                pos += 1
                state = 1
            elif c in _DIGIT_CHARS:
                pos += 1
                state = 1
            else:
                raise JSONParseError(f"bad number at {pos}")
        elif state == 1:
            if c in _DIGIT_CHARS:
                pos += 1
            elif c == ".":
                pos += 1
                state = 2
            elif c in ("e", "E"):
                pos += 1
                state = 3
            else:
                break
        elif state == 2:
            if c in _DIGIT_CHARS:
                pos += 1
            elif c in ("e", "E"):
                pos += 1
                state = 3
            else:
                break
        elif state == 3:
            if c in ("+", "-"):
                pos += 1
            elif c in _DIGIT_CHARS:
                pos += 1
            else:
                raise JSONParseError(f"bad number exponent at {pos}")
    s = text[start:pos]
    # Determine if it's float or int based on presence of decimal or exponent
    is_float = ("." in s) or ("e" in s.lower())
    if is_float:
        return float(s), pos
    return int(s), pos
