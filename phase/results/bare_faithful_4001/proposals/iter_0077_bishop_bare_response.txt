def _skip_ws(text: str, pos: int) -> int:
    # State machine for whitespace skipping
    while pos < len(text):
        c = text[pos]
        if c == ' ':
            pos += 1
        elif c == '\t':
            pos += 1
        elif c == '\n':
            pos += 1
        elif c == '\r':
            pos += 1
        else:
            break
    return pos
