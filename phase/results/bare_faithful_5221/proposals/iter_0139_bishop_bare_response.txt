def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    # DFA-based number parsing
    n = len(text)
    start = pos
    state = 0  # 0: start, 1: int, 2: frac, 3: exp, 4: exp_sign
    while pos < n:
        c = text[pos]
        if state == 0:  # start
            if c == "-":
                pos += 1
            state = 1
        elif state == 1:  # int
            if c == "0":
                pos += 1
            elif c in _DIGIT_CHARS:
                pos += 1
            elif c == ".":
                state = 2
                pos += 1
            elif c in ("e", "E"):
                state = 3
                pos += 1
            else:
                break
        elif state == 2:  # frac
            if c in _DIGIT_CHARS:
                pos += 1
            elif c in ("e", "E"):
                state = 3
                pos += 1
            else:
                break
        elif state == 3:  # exp
            if c in ("+", "-"):
                state = 4
                pos += 1
            elif c in _DIGIT_CHARS:
                pos += 1
            else:
                break
        elif state == 4:  # exp_sign
            if c in _DIGIT_CHARS:
                pos += 1
            else:
                break
    if pos == start:
        raise JSONParseError(f"bad number at {pos}")
    s = text[start:pos]
    # Check if it's a float by looking for '.' or 'e'
    is_float = "." in s or "e" in s.lower()
    if is_float:
        return float(s), pos
    return int(s), pos
