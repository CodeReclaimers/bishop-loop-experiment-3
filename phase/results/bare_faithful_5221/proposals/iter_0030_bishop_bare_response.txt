import re

_NUMBER_RE = re.compile(r"""
    -? # optional minus
    (?:0|[1-9]\d*) # integer part: 0 or non-zero digit then digits
    (?:\.\d+)? # optional fractional part
    (?:[eE][+-]?\d+)? # optional exponent part
""", re.VERBOSE)

def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    match = _NUMBER_RE.match(text, pos)
    if not match:
        raise JSONParseError(f"bad number at {pos}")
    s = match.group(0)
    end_pos = pos + len(s)
    # Use float conversion for all numbers to handle both int and float cases
    # This is more efficient than checking for decimal point and exponent
    # and avoids repeated character-by-character parsing
    try:
        if '.' in s or 'e' in s.lower():
            return float(s), end_pos
        else:
            return int(s), end_pos
    except ValueError:
        raise JSONParseError(f"bad number at {pos}")
