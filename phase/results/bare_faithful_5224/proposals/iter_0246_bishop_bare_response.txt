def _is_hex4(s: str) -> bool:
    if len(s) != 4:
        return False
    for ch in s:
        if not (("0" <= ch <= "9") or ("a" <= ch <= "f") or ("A" <= ch <= "F")):
            return False
    return True


def _hex4_to_int(s: str) -> int:
    """Convert a 4-character hexadecimal string to an integer directly."""
    result = 0
    for ch in s:
        if "0" <= ch <= "9":
            digit = ord(ch) - ord("0")
        elif "a" <= ch <= "f":
            digit = ord(ch) - ord("a") + 10
        elif "A" <= ch <= "F":
            digit = ord(ch) - ord("A") + 10
        else:
            raise ValueError(f"Invalid hex digit: {ch}")
        result = result * 16 + digit
    return result
