def _skip_ws(text: str, pos: int) -> int:
    n = len(text)
    # Skip whitespace in chunks of 8 characters for better performance
    while pos <= n - 8:
        chunk = text[pos:pos + 8]
        if not all(c in _WS_CHARS for c in chunk):
            break
        pos += 8
    while pos < n and text[pos] in _WS_CHARS:
        pos += 1
    return pos
