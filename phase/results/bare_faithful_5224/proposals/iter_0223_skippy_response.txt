def _skip_ws(text: str, pos: int) -> int:
    # Use a regex for whitespace skipping, which is faster than character-by-character
    import re
    m = re.match(r'\s*', text[pos:])
    if m is None:
        return pos  # Should not happen, but just in case
    return pos + len(m.group(0))
