def _skip_ws(text: str, pos: int) -> int:
    # Convert to bytes for faster scanning
    text_bytes = text.encode('utf-8')
    n = len(text_bytes)
    while pos < n:
        byte_val = text_bytes[pos]
        if byte_val in (32, 9, 10, 13):  # space, tab, newline, carriage return
            pos += 1
        else:
            break
    return pos
