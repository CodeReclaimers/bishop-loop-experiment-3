def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    # Use regex to match the number and then convert to int/float
    import re
    # Match a number (integer or float) including scientific notation
    match = re.match(r'[-+]?(?:\d+\.?\d*|\.\d+)(?:[eE][+-]?\d+)?', text[pos:])
    if not match:
        raise JSONParseError(f"bad number at {pos}")
    s = match.group(0)
    end_pos = pos + len(s)
    # Try int first, then float if that fails
    try:
        return int(s), end_pos
    except ValueError:
        return float(s), end_pos
