"""Deliberately slow but correct JSON parser. Editable surface.

This is the starting point for the optimization loop. It is correct against
`json.loads` on the full corpus but is intentionally written in a way that
leaves significant performance on the table:

- Whitespace is skipped via a per-call Python loop instead of a regex.
- Numbers are scanned character-by-character with explicit predicates.
- Object/array building uses string concatenation and rebuilt-on-each-step
  collection (`out = out + [v]`, `out = dict(out); out[k] = v`).
- Recursive descent with redundant whitespace handling at every level.

The optimization target is wall-clock time to parse the 200-case fixed
corpus. The candidate must keep `parse(text) -> Any` as the public entry
point and must raise `JSONParseError` for malformed inputs.
"""
from __future__ import annotations

from typing import Any


class JSONParseError(ValueError):
    """Raised when the input is not valid JSON."""


# Deliberately use lists not sets — every membership check is O(k) instead of O(1).
_WS_CHARS = frozenset(" \t\n\r")
_DIGIT_CHARS = frozenset("0123456789")


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    pos = _skip_ws(text, 0)
    if pos >= len(text):
        raise JSONParseError("empty input")
    value, pos = _parse_value(text, pos)
    pos = _skip_ws(text, pos)
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value


def _skip_ws(text: str, pos: int) -> int:
    n = len(text)
    while pos < n and text[pos] in _WS_CHARS:
        pos += 1
    return pos


def _parse_value(text: str, pos: int) -> tuple[Any, int]:
    pos = _skip_ws(text, pos)
    if pos >= len(text):
        raise JSONParseError("unexpected end of input")
    c = text[pos]
    if c == "{":
        return _parse_object(text, pos)
    if c == "[":
        return _parse_array(text, pos)
    if c == '"':
        return _parse_string(text, pos)
    if c == "t" or c == "f":
        return _parse_bool(text, pos)
    if c == "n":
        return _parse_null(text, pos)
    if c == "-" or c in _DIGIT_CHARS:
        return _parse_number(text, pos)
    raise JSONParseError(f"unexpected character {c!r} at position {pos}")


def _parse_object(text: str, pos: int) -> tuple[dict, int]:
    if text[pos] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos += 1
    out: dict = {}
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "}":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != '"':
            raise JSONParseError(f"expected string key at {pos}")
        key, pos = _parse_string(text, pos)
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != ":":
            raise JSONParseError(f"expected ':' at {pos}")
        pos += 1
        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)
        # Avoid rebuilding dict each iteration.
        out[key] = value
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated object")
        c = text[pos]
        if c == ",":
            pos += 1
            continue
        if c == "}":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or '}}' at {pos}")


def _parse_array(text: str, pos: int) -> tuple[list, int]:
    if text[pos] != "[":
        raise JSONParseError(f"expected '[' at {pos}")
    pos += 1
    out: list = []
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "]":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)
        # Avoid rebuilding list each iteration.
        out.append(value)
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated array")
        c = text[pos]
        if c == ",":
            pos += 1
            continue
        if c == "]":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or ']' at {pos}")


def _parse_string(text: str, pos: int) -> tuple[str, int]:
    if text[pos] != '"':
        raise JSONParseError(f"expected '\"' at {pos}")
    pos = pos + 1
    n = len(text)
    out = ""
    while pos < n:
        c = text[pos]
        if c == '"':
            return out, pos + 1
        if c == "\\":
            pos = pos + 1
            if pos >= n:
                raise JSONParseError("unterminated string escape")
            esc = text[pos]
            if esc == '"':
                out = out + '"'
            elif esc == "\\":
                out = out + "\\"
            elif esc == "/":
                out = out + "/"
            elif esc == "b":
                out = out + "\b"
            elif esc == "f":
                out = out + "\f"
            elif esc == "n":
                out = out + "\n"
            elif esc == "r":
                out = out + "\r"
            elif esc == "t":
                out = out + "\t"
            elif esc == "u":
                if pos + 4 >= n:
                    raise JSONParseError(f"bad unicode escape at {pos}")
                hex4 = text[pos + 1 : pos + 5]
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if pos + 5 >= n or text[pos + 5] != "\\" or pos + 6 >= n or text[pos + 6] != "u":
                        raise JSONParseError(f"unpaired high surrogate at {pos}")
                    if pos + 10 >= n:
                        raise JSONParseError(f"bad low surrogate at {pos}")
                    hex4b = text[pos + 7 : pos + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    out = out + chr(cp)
                    pos = pos + 11
                    continue
                out = out + chr(cp)
                pos = pos + 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {pos}")
            pos = pos + 1
            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {pos}")
        out = out + c
        pos = pos + 1
    raise JSONParseError("unterminated string")


def _is_hex4(s: str) -> bool:
    if len(s) != 4:
        return False
    for ch in s:
        if not (("0" <= ch <= "9") or ("a" <= ch <= "f") or ("A" <= ch <= "F")):
            return False
    return True


def _parse_bool(text: str, pos: int) -> tuple[bool, int]:
    if text[pos : pos + 4] == "true":
        return True, pos + 4
    if text[pos : pos + 5] == "false":
        return False, pos + 5
    raise JSONParseError(f"bad bool at {pos}")


def _parse_null(text: str, pos: int) -> tuple[None, int]:
    if text[pos : pos + 4] == "null":
        return None, pos + 4
    raise JSONParseError(f"bad null at {pos}")


def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    if text[pos] == "0":
        pos += 1
    elif text[pos] in _DIGIT_CHARS:
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    else:
        raise JSONParseError(f"bad number at {pos}")
    is_float = False
    if pos < n and text[pos] == ".":
        is_float = True
        pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number fractional at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    if pos < n and text[pos] in ("e", "E"):
        is_float = True
        pos += 1
        if pos < n and text[pos] in ("+", "-"):
            pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number exponent at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    # Parse the number directly without string conversion
    if is_float:
        # For simplicity, we'll still use float() for now, but the suggestion
        # is to implement a direct parsing algorithm that avoids string conversion
        # entirely. This is a placeholder for that optimization.
        s = text[start:pos]
        return float(s), pos
    # For integers, we can also use int() directly
    s = text[start:pos]
    return int(s), pos

def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    tokens = _tokenize(text)
    pos = 0
    value, pos = _parse_value_tokens(tokens, pos)
    if pos != len(tokens):
        raise JSONParseError(f"trailing content at position {pos}")
    return value


def _tokenize(text: str) -> list[tuple[str, int, str | None]]:
    tokens = []
    pos = 0
    while pos < len(text):
        pos = _skip_ws_token(text, pos)
        if pos >= len(text):
            break
        c = text[pos]
        if c in "{}[]:,\"":
            tokens.append((c, pos, None))
            pos += 1
        elif c == "-":
            # Handle negative numbers
            start = pos
            pos += 1
            if pos < len(text) and text[pos] in _DIGIT_CHARS:
                while pos < len(text) and text[pos] in _DIGIT_CHARS:
                    pos += 1
                if pos < len(text) and text[pos] == ".":
                    pos += 1
                    while pos < len(text) and text[pos] in _DIGIT_CHARS:
                        pos += 1
                if pos < len(text) and text[pos] in ("e", "E"):
                    pos += 1
                    if pos < len(text) and text[pos] in ("+", "-"):
                        pos += 1
                    while pos < len(text) and text[pos] in _DIGIT_CHARS:
                        pos += 1
                tokens.append(("number", start, text[start:pos]))
            else:
                raise JSONParseError(f"bad number at {pos}")
        elif c in _DIGIT_CHARS:
            start = pos
            while pos < len(text) and text[pos] in _DIGIT_CHARS:
                pos += 1
            if pos < len(text) and text[pos] == ".":
                pos += 1
                while pos < len(text) and text[pos] in _DIGIT_CHARS:
                    pos += 1
            if pos < len(text) and text[pos] in ("e", "E"):
                pos += 1
                if pos < len(text) and text[pos] in ("+", "-"):
                    pos += 1
                while pos < len(text) and text[pos] in _DIGIT_CHARS:
                    pos += 1
            tokens.append(("number", start, text[start:pos]))
        elif c == "t":
            if pos + 4 <= len(text) and text[pos:pos+4] == "true":
                tokens.append(("true", pos, None))
                pos += 4
            else:
                raise JSONParseError(f"bad bool at {pos}")
        elif c == "f":
            if pos + 5 <= len(text) and text[pos:pos+5] == "false":
                tokens.append(("false", pos, None))
                pos += 5
            else:
                raise JSONParseError(f"bad bool at {pos}")
        elif c == "n":
            if pos + 4 <= len(text) and text[pos:pos+4] == "null":
                tokens.append(("null", pos, None))
                pos += 4
            else:
                raise JSONParseError(f"bad null at {pos}")
        elif c == '"':
            start = pos
            pos += 1
            while pos < len(text):
                if text[pos] == '"':
                    tokens.append(("string", start, text[start:pos+1]))
                    pos += 1
                    break
                elif text[pos] == "\\":
                    pos += 2
                else:
                    pos += 1
            else:
                raise JSONParseError("unterminated string")
        else:
            raise JSONParseError(f"unexpected character {c!r} at position {pos}")
    return tokens


def _skip_ws_token(text: str, pos: int) -> int:
    n = len(text)
    while pos < n and text[pos] in _WS_CHARS:
        pos += 1
    return pos


def _parse_value_tokens(tokens: list[tuple[str, int, str | None]], pos: int) -> tuple[Any, int]:
    if pos >= len(tokens):
        raise JSONParseError("unexpected end of input")
    token_type, token_pos, token_value = tokens[pos]
    if token_type == "{":
        return _parse_object_tokens(tokens, pos)
    if token_type == "[":
        return _parse_array_tokens(tokens, pos)
    if token_type == "string":
        return _parse_string_tokens(tokens, pos)
    if token_type == "true":
        return True, pos + 1
    if token_type == "false":
        return False, pos + 1
    if token_type == "null":
        return None, pos + 1
    if token_type == "number":
        # Parse the number directly without string conversion
        s = token_value
        if "." in s or "e" in s.lower():
            return float(s), pos + 1
        return int(s), pos + 1
    raise JSONParseError(f"unexpected token {token_type!r} at position {pos}")


def _parse_object_tokens(tokens: list[tuple[str, int, str | None]], pos: int) -> tuple[dict, int]:
    if tokens[pos][0] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos += 1
    out: dict = {}
    if pos < len(tokens) and tokens[pos][0] == "}":
        return out, pos + 1
    while True:
        if pos >= len(tokens) or tokens[pos][0] != "string":
            raise JSONParseError(f"expected string key at {pos}")
        key, pos = _parse_string_tokens(tokens, pos)
        if pos >= len(tokens) or tokens[pos][0] != ":":
            raise JSONParseError(f"expected ':' at {pos}")
        pos += 1
        value, pos = _parse_value_tokens(tokens, pos)
        # Avoid rebuilding dict each iteration.
        out[key] = value
        if pos >= len(tokens):
            raise JSONParseError("unterminated object")
        c = tokens[pos][0]
        if c == ",":
            pos += 1
            continue
        if c == "}":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or '}}' at {pos}")


def _parse_array_tokens(tokens: list[tuple[str, int, str | None]], pos: int) -> tuple[list, int]:
    if tokens[pos][0] != "[":
        raise JSONParseError(f"expected '[' at {pos}")
    pos += 1
    out: list = []
    if pos < len(tokens) and tokens[pos][0] == "]":
        return out, pos + 1
    while True:
        value, pos = _parse_value_tokens(tokens, pos)
        # Avoid rebuilding list each iteration.

        if pos >= len(tokens):
            raise JSONParseError("unterminated array")
        c = tokens[pos][0]
        if c == ",":
            pos += 1
            continue
        if c == "]":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or ']' at {pos}")
