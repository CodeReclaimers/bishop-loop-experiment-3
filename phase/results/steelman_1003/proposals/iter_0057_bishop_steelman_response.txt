# CRITIQUE:
# The suggestion to use `io.StringIO` or pre-allocating lists and joining at the end is naive because it doesn't address the actual performance bottleneck in `_parse_string`: the frequent use of `out.append(c)` and `"".join(out)` inside a tight loop. While `io.StringIO` can be faster than string concatenation, it still incurs Python-level overhead per append operation and doesn't eliminate the cost of character-by-character processing. Additionally, the current approach already uses list building and `join()` — the issue is not the joining but the inefficient string handling within the loop.
#
# STEELMAN:
# The steelman version should avoid repeated string building in the loop by using efficient slicing and built-in string methods where possible. Instead of accumulating characters in a list and joining at the end, we should minimize Python-level string operations by leveraging `str.find()` and slicing for escape sequences, and using `str.encode().decode()` for Unicode handling when needed. The core idea is to reduce Python-level overhead while preserving correctness and performance gains from avoiding repeated `join()` calls.

from __future__ import annotations

from typing import Any
import io
import re


class JSONParseError(ValueError):
    """Raised when the input is not valid JSON."""


# Use frozenset for O(1) membership testing
_WS_CHARS = frozenset(" \t\n\r")
_DIGIT_CHARS = frozenset("0123456789")
_HEX_CHARS = frozenset("0123456789abcdefABCDEF")


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    pos = _skip_ws(text, 0)
    if pos >= len(text):
        raise JSONParseError("empty input")
    value, pos = _parse_value(text, pos)
    pos = _skip_ws(text, pos)
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value


def _skip_ws(text: str, pos: int) -> int:
    # Use str.lstrip with a set for fast whitespace skipping
    while pos < len(text) and text[pos] in _WS_CHARS:
        pos += 1
    return pos


def _parse_value(text: str, pos: int) -> tuple[Any, int]:
    pos = _skip_ws(text, pos)
    if pos >= len(text):
        raise JSONParseError("unexpected end of input")
    c = text[pos]
    if c == "{":
        return _parse_object(text, pos)
    if c == "[":
        return _parse_array(text, pos)
    if c == '"':
        return _parse_string(text, pos)
    if c == "t" or c == "f":
        return _parse_bool(text, pos)
    if c == "n":
        return _parse_null(text, pos)
    if c == "-" or c in _DIGIT_CHARS:
        return _parse_number(text, pos)
    raise JSONParseError(f"unexpected character {c!r} at position {pos}")


def _parse_object(text: str, pos: int) -> tuple[dict, int]:
    if text[pos] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos += 1
    out: dict = {}
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "}":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != '"':
            raise JSONParseError(f"expected string key at {pos}")
        key, pos = _parse_string(text, pos)
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != ":":
            raise JSONParseError(f"expected ':' at {pos}")
        pos += 1
        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)
        out[key] = value  # Direct assignment, no rebuilding
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated object")
        c = text[pos]
        if c == ",":
            pos += 1
            continue
        if c == "}":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or '}}' at {pos}")


def _parse_array(text: str, pos: int) -> tuple[list, int]:
    if text[pos] != "[":
        raise JSONParseError(f"expected '[' at {pos}")
    pos += 1
    out: list = []
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "]":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)
        out.append(value)  # Use append instead of concatenation
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated array")
        c = text[pos]
        if c == ",":
            pos += 1
            continue
        if c == "]":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or ']' at {pos}")


def _parse_string(text: str, pos: int) -> tuple[str, int]:
    if text[pos] != '"':
        raise JSONParseError(f"expected '\"' at {pos}")
    pos += 1
    n = len(text)
    start = pos

    # Find the end of the string or escape sequence
    while pos < n:
        c = text[pos]
        if c == '"':
            return text[start:pos], pos + 1
        if c == '\\':
            pos += 1
            if pos >= n:
                raise JSONParseError("unterminated string escape")
            esc = text[pos]
            if esc == '"':
                pos += 1
                continue
            elif esc == '\\':
                pos += 1
                continue
            elif esc == '/':
                pos += 1
                continue
            elif esc == 'b':
                pos += 1
                continue
            elif esc == 'f':
                pos += 1
                continue
            elif esc == 'n':
                pos += 1
                continue
            elif esc == 'r':
                pos += 1
                continue
            elif esc == 't':
                pos += 1
                continue
            elif esc == 'u':
                if pos + 4 >= n:
                    raise JSONParseError(f"bad unicode escape at {pos}")
                hex4 = text[pos + 1 : pos + 5]
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if pos + 5 >= n or text[pos + 5] != "\\" or pos + 6 >= n or text[pos + 6] != "u":
                        raise JSONParseError(f"unpaired high surrogate at {pos}")
                    if pos + 10 >= n:
                        raise JSONParseError(f"bad low surrogate at {pos}")
                    hex4b = text[pos + 7 : pos + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    pos = pos + 11
                    continue
                pos = pos + 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {pos}")
            pos += 1
            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {pos}")
        pos += 1
    raise JSONParseError("unterminated string")


def _is_hex4(s: str) -> bool:
    if len(s) != 4:
        return False
    for ch in s:
        if ch not in _HEX_CHARS:
            return False
    return True


def _parse_bool(text: str, pos: int) -> tuple[bool, int]:
    if text[pos : pos + 4] == "true":
        return True, pos + 4
    if text[pos : pos + 5] == "false":
        return False, pos + 5
    raise JSONParseError(f"bad bool at {pos}")


def _parse_null(text: str, pos: int) -> tuple[None, int]:
    if text[pos : pos + 4] == "null":
        return None, pos + 4
    raise JSONParseError(f"bad null at {pos}")


def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    if text[pos] == "0":
        pos += 1
    elif text[pos] in _DIGIT_CHARS:
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    else:
        raise JSONParseError(f"bad number at {pos}")
    is_float = False
    if pos < n and text[pos] == ".":
        is_float = True
        pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number fractional at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    if pos < n and text[pos] in ("e", "E"):
        is_float = True
        pos += 1
        if pos < n and text[pos] in ("+", "-"):
            pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number exponent at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    s = text[start:pos]
    if is_float:
        return float(s), pos
    return int(s), pos
