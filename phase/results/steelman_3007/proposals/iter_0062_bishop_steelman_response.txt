CRITIQUE: The suggestion to preprocess all whitespace with a regex and then manually track positions fails because it destroys the ability to report accurate error positions — the parser would have no way to map cleaned positions back to original text locations. Additionally, the regex substitution approach doesn't handle escaped quotes or other structural elements that must be preserved for correct parsing.
STEELMAN: A better approach is to preprocess only the whitespace, keeping a mapping from cleaned positions back to original positions, and then use this mapping during parsing to maintain accurate error reporting while still benefiting from bulk whitespace skipping.

- Using regex for whitespace skipping

    pos = _skip_ws(text, pos)
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value


def _skip_ws(text: str, pos: int) -> int:

    while pos < len(text) and text[pos] in _WS_CHARS:
        pos += 1
    return pos


def _parse_value(text: str, pos: int) -> tuple[Any, int]:

        raise JSONParseError("unexpected end of input")
    c = text[pos]

    if c == "[":

        raise JSONParseError(f"expected '\"' at {pos}")

                out.append('"')

                out.append("/")

                out.append("\b")
            elif esc == "f":

                pos = pos + 4

                raise JSONParseError(f"bad escape \\{esc} at {pos}")

            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {pos}")
        out.append(c)
        pos = pos + 1
    raise JSONParseError("unterminated string")

    for ch in s:

    if text[pos : pos + 4] == "true":
        return True, pos + 4
    if text[pos : pos + 5] == "false":
