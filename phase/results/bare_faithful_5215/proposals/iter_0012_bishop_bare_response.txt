    n = len(text)
    start = pos
    # State 0: start of number
    # State 1: after minus sign
    # State 2: integer part
    # State 3: decimal point
    # State 4: fractional part
    # State 5: exponent start
    # State 6: exponent sign
    # State 7: exponent part
    state = 0
    while pos < n:
        c = text[pos]
        if state == 0:
            if c == "-":
                state = 1
            elif c == "0":
                state = 2
            elif c in _DIGIT_CHARS:
                state = 2
            else:
                raise JSONParseError(f"bad number at {pos}")

            if c == "0":
                state = 2
            elif c in _DIGIT_CHARS:
                state = 2
