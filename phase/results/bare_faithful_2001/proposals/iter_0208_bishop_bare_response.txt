    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:

    if text[pos] == "0":
        pos += 1
    elif text[pos] in _DIGIT_CHARS:
        # Find end of integer part using str.find
        int_end = pos
        while int_end < n and text[int_end] in _DIGIT_CHARS:
            int_end += 1
        pos = int_end
    else:
        raise JSONParseError(f"bad number at {pos}")

    decimal_pos = n
