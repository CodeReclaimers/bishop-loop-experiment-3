def _skip_ws(text: str, pos: int) -> int:
    # Use C-level str.find to locate next whitespace character
    while pos < len(text):
        found = False
        for ws_char in _WS_CHARS:
            next_pos = text.find(ws_char, pos)
            if next_pos != -1:
                if not found or next_pos < found:
                    found = next_pos
        if found:
            pos = found
            break
        pos += 1
    return pos
