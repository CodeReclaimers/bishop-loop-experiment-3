def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    # Find the end of the number using text.find for bulk scanning
    # Look for the first character that is not part of a number
    end_pos = pos
    # Scan digits (including leading zero)
    while end_pos < n and text[end_pos] in _DIGIT_CHARS:
        end_pos += 1
    # Check for fractional part
    if end_pos < n and text[end_pos] == ".":
        end_pos += 1
        while end_pos < n and text[end_pos] in _DIGIT_CHARS:
            end_pos += 1
    # Check for exponent
    if end_pos < n and text[end_pos] in ("e", "E"):
        end_pos += 1
        if end_pos < n and text[end_pos] in ("+", "-"):
            end_pos += 1
        while end_pos < n and text[end_pos] in _DIGIT_CHARS:
            end_pos += 1
    pos = end_pos
    s = text[start:pos]
    # Determine if it's a float or int
    is_float = "." in s or "e" in s.lower()
    if is_float:
        return float(s), pos
    return int(s), pos
