def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    # Single loop to parse the entire number
    while pos < n:
        c = text[pos]
        if c in _DIGIT_CHARS or c == "." or c in "eE" or c in "+-":
            pos += 1
        else:
            break
    # Validate and parse the number
    s = text[start:pos]
    if not s:
        raise JSONParseError(f"bad number at {start}")
    # Check for valid number format
    if s.count('.') > 1 or s.count('e') > 1 or s.count('E') > 1:
        raise JSONParseError(f"bad number at {start}")
    # Determine if it's a float or int
    is_float = '.' in s or 'e' in s.lower()
    if is_float:
        return float(s), pos
    return int(s), pos
