    n = len(text)

        if c == '"':
            return out, pos + 1

            pos = pos + 1
            if pos >= n:
                raise JSONParseError("unterminated string escape")
            esc = text[pos]
            if esc == '"':
                out = out + '"'
            elif esc == "\\":
                out = out + "\\"
            elif esc == "/":
                out = out + "/"
            elif esc == "b":
                out = out + "\b"
            elif esc == "f":
                out = out + "\f"

                out = out + "\n"
            elif esc == "r":
                out = out + "\r"
            elif esc == "t":
                out = out + "\t"

                    raise JSONParseError(f"bad unicode escape at {pos}")
