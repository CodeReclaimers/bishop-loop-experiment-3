def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    import re
    # Match integer or float number
    m = re.match(r'[-+]?(?:\d+\.?\d*|\.\d+)(?:[eE][+-]?\d+)?', text[pos:])
    if not m:
        raise JSONParseError(f"bad number at {pos}")
    s = m.group(0)
    pos += len(s)
    if '.' in s or 'e' in s.lower():
        return float(s), pos
    return int(s), pos
