def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    # Single loop to consume all digits, decimal points, and exponents
    while pos < n:
        c = text[pos]
        if c in _DIGIT_CHARS or c == "." or c in "eE" or c in "+-":
            if c == ".":
                if pos + 1 < n and text[pos + 1] not in _DIGIT_CHARS:
                    raise JSONParseError(f"bad number fractional at {pos}")
            elif c in "eE":
                if pos + 1 < n and text[pos + 1] in "+-":
                    pos += 1
                if pos + 1 >= n or text[pos + 1] not in _DIGIT_CHARS:
                    raise JSONParseError(f"bad number exponent at {pos}")
            pos += 1
        else:
            break
    # Now we have the complete number string
    s = text[start:pos]
    # Check if it's a float by looking for decimal point or exponent
    is_float = "." in s or "e" in s.lower()
    if is_float:
        return float(s), pos
    return int(s), pos
