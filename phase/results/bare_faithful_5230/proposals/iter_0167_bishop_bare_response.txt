def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    # State machine for number parsing
    state = 0  # 0: start, 1: integer part, 2: decimal point, 3: fractional part, 4: exponent start, 5: exponent part
    while pos < n:
        c = text[pos]
        if state == 0:
            if c == "-":
                pos += 1
            state = 1
        elif state == 1:
            if c == "0":
                pos += 1
                state = 2
            elif c in _DIGIT_CHARS:
                pos += 1
            else:
                raise JSONParseError(f"bad number at {pos}")
        elif state == 2:
            if c == ".":
                pos += 1
                state = 3
            elif c in ("e", "E"):
                pos += 1
                state = 4
            else:
                break
        elif state == 3:
            if c in _DIGIT_CHARS:
                pos += 1
            else:
                raise JSONParseError(f"bad number fractional at {pos}")
        elif state == 4:
            if c in ("+", "-"):
                pos += 1
            state = 5
        elif state == 5:
            if c in _DIGIT_CHARS:
                pos += 1
            else:
                break
    s = text[start:pos]
    # Determine if it's a float or int
    is_float = '.' in s or 'e' in s.lower()
    if is_float:
        return float(s), pos
    return int(s), pos
