def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    pos = 0
    while pos < len(text):
        # Skip any leading whitespace
        while pos < len(text) and text[pos] in _WS_CHARS:
            pos += 1
        
        # Parse the next token
        if pos >= len(text):
            raise JSONParseError("unexpected end of input")
        c = text[pos]
        if c == "{":
            value, pos = _parse_object(text, pos)
        elif c == "[":
            value, pos = _parse_array(text, pos)
        elif c == '"':
            value, pos = _parse_string(text, pos)
        elif c == "t" or c == "f":
            value, pos = _parse_bool(text, pos)
        elif c == "n":
            value, pos = _parse_null(text, pos)
        elif c == "-" or c in _DIGIT_CHARS:
            value, pos = _parse_number(text, pos)
        else:
            # If no valid token was found, raise an error
            raise JSONParseError(f"unexpected character {c!r} at position {pos}")
    
    # Skip any trailing whitespace
    while pos < len(text) and text[pos] in _WS_CHARS:
        pos += 1
    
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value
