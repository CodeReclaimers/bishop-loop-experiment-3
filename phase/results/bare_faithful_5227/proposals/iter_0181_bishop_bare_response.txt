def _skip_ws(text: str, pos: int) -> int:
    # State machine for whitespace skipping
    state = "start"
    while pos < len(text):
        c = text[pos]
        if state == "start":
            if c in _WS_CHARS:
                pos += 1
                continue
            else:
                state = "done"
        elif state == "done":
            break
    return pos
