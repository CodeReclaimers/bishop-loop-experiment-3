def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    if text[pos] == "0":
        pos += 1
    elif text[pos] in _DIGIT_CHARS:
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    else:
        raise JSONParseError(f"bad number at {pos}")
    is_float = False
    # Directly scan for decimal point and exponent to avoid repeated scanning
    decimal_pos = -1
    exp_pos = -1
    i = pos
    while i < n:
        c = text[i]
        if c == ".":
            decimal_pos = i
            is_float = True
        elif c in ("e", "E"):
            exp_pos = i
            is_float = True
        elif c not in _DIGIT_CHARS:
            break
        i += 1
    pos = i
    if decimal_pos != -1:
        # Skip the decimal point
        pos = decimal_pos + 1
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    if exp_pos != -1:
        # Skip the exponent indicator
        pos = exp_pos + 1
        if pos < n and text[pos] in ("+", "-"):
            pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number exponent at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    s = text[start:pos]
    if is_float:
        return float(s), pos
    return int(s), pos
