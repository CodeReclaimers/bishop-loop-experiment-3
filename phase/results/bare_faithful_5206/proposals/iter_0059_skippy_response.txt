def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    # Use regex to parse number, much faster than character-by-character
    # Match integer or float, including scientific notation
    m = re.match(r'(-?(?:0|[1-9]\d*)(?:\.\d+)?(?:[eE][+-]?\d+)?)', text[pos:])
    if not m:
        raise JSONParseError(f"bad number at {pos}")
    s = m.group(1)
    pos += len(s)
    # Try to parse as int first, then float
    try:
        return int(s), pos
    except ValueError:
        return float(s), pos
