def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    import re
    # Match complete number pattern including scientific notation
    match = re.match(r'(-?(?:0|[1-9]\d*)(?:\.\d+)?(?:[eE][+-]?\d+)?)', text[pos:])
    if not match:
