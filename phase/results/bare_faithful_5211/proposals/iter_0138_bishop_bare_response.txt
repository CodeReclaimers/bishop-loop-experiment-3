def _skip_ws(text: str, pos: int) -> int:
    n = len(text)
    while pos < n and text[pos] in _WS_CHARS:
        pos += 1
    return pos


def _preprocess_whitespace(text: str) -> str:
    """Remove unnecessary whitespace from the input text."""
    result = []
    i = 0
    while i < len(text):
        char = text[i]
        if char in _WS_CHARS:
            # Skip whitespace characters
            i += 1
        else:
            result.append(char)
            i += 1
    return ''.join(result)
