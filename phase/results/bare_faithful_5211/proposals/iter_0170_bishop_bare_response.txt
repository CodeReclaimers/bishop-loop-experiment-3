def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    if text[pos] == "0":
        pos += 1
    elif text[pos] in _DIGIT_CHARS:
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    else:
        raise JSONParseError(f"bad number at {pos}")
    is_float = False
    if pos < n and text[pos] == ".":
        is_float = True
        pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number fractional at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    if pos < n and text[pos] in ("e", "E"):
        is_float = True
        pos += 1
        if pos < n and text[pos] in ("+", "-"):
            pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number exponent at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    if is_float:
        return float(text[start:pos]), pos
    # Use int() for large integers to avoid Python's arbitrary precision
    # but still handle the conversion efficiently
    num_str = text[start:pos]
    try:
        # Try to parse as int first, fallback to float if needed
        if '.' in num_str or 'e' in num_str.lower():
            return float(num_str), pos
        return int(num_str), pos
    except ValueError:
        return float(num_str), pos
