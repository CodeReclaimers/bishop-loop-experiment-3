def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    # State machine for number parsing
    state = 0  # 0: start, 1: int, 2: frac, 3: exp, 4: exp_sign
    while pos < n:
        c = text[pos]
        if state == 0:  # start
            if c == "-":
                pos += 1
            state = 1
        elif state == 1:  # int
            if c == "0":
                pos += 1
            elif c in _DIGIT_CHARS:
                pos += 1
            else:
                state = 2 if c == "." else 3 if c in "eE" else 5
                continue
        elif state == 2:  # frac
            if c == ".":
                pos += 1
            elif c in _DIGIT_CHARS:
                pos += 1
            else:
                state = 3 if c in "eE" else 5
                continue
        elif state == 3:  # exp
            if c in "eE":
                pos += 1
            elif c in "+-":
                pos += 1
                state = 4
            elif c in _DIGIT_CHARS:
                pos += 1
            else:
                state = 5
                continue
        elif state == 4:  # exp_sign
            if c in _DIGIT_CHARS:
                pos += 1
            else:
                state = 5
                continue
        else:  # final state
            break
    s = text[start:pos]
    # Check if it's a float or int
    is_float = ("." in s) or ("e" in s.lower())
    if is_float:
        return float(s), pos
    return int(s), pos
