# Precompile the regex for whitespace skipping
_WS_REGEX = re.compile(r'^[ \t\n\r]*')

def _skip_ws(text: str, pos: int) -> int:
    # Optimized inline whitespace skipping for small inputs
    # Use regex for larger inputs to reduce loop overhead
    if pos < 1000:  # Threshold for when regex becomes beneficial
        try:
            while text[pos] in _WS_CHARS:
                pos += 1
        except IndexError:
            pass
    else:
        # Use regex for large inputs
