import re

# Pre-compile a regex for skipping large chunks of whitespace
_WS_REGEX = re.compile(r'^[ \t\n\r]*')

def _skip_ws(text: str, pos: int) -> int:
    # Optimized inline whitespace skipping for small runs
    try:
        while text[pos] in _WS_CHARS:
            pos += 1
    except IndexError:
        pass
    else:
        # For larger runs, use regex to skip efficiently
        match = _WS_REGEX.match(text, pos)
        if match:
            pos += len(match.group(0))
    return pos
