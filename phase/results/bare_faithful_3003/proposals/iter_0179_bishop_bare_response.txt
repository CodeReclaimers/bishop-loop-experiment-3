def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    # Precompute non-whitespace character positions
    non_ws_indices = []
    for i, char in enumerate(text):
        if char not in _WS_CHARS:
            non_ws_indices.append(i)
    
    pos = 0
    if not non_ws_indices:
        raise JSONParseError("empty input")
    
    # Skip initial whitespace
    while pos < len(non_ws_indices) and non_ws_indices[pos] == 0:
        pos += 1
    if pos >= len(non_ws_indices):
        raise JSONParseError("empty input")
    
    # Use binary search to find next non-whitespace position
    def _skip_ws(pos: int) -> int:
        # Binary search for the first index >= pos
        left, right = 0, len(non_ws_indices) - 1
        while left <= right:
            mid = (left + right) // 2
            if non_ws_indices[mid] >= pos:
                right = mid - 1
            else:
                left = mid + 1
