def _skip_ws(text: str, pos: int) -> int:
    # Binary search for the next non-whitespace character
    n = len(text)
    if pos >= n:
        return pos
    # Precompute non-whitespace character positions
    if not hasattr(_skip_ws, 'non_ws_positions'):
        _skip_ws.non_ws_positions = [i for i in range(n) if text[i] not in _WS_CHARS]
    # Binary search
    left, right = 0, len(_skip_ws.non_ws_positions) - 1
    while left <= right:
        mid = (left + right) // 2
        idx = _skip_ws.non_ws_positions[mid]
        if idx >= pos:
            right = mid - 1
        else:
