def _skip_ws(text: str, pos: int) -> int:
    # Binary search for the next non-whitespace character
    # Precompute non-whitespace indices once per parse call
    if not hasattr(_skip_ws, 'non_ws_indices'):
        _skip_ws.non_ws_indices = [i for i in range(len(text)) if text[i] not in _WS_CHARS]
    
    # Binary search for the first non-whitespace index >= pos
    left, right = 0, len(_skip_ws.non_ws_indices) - 1
    result = len(text)
    while left <= right:
        mid = (left + right) // 2
        idx = _skip_ws.non_ws_indices[mid]
        if idx >= pos:
            result = idx
            right = mid - 1
        else:
            left = mid + 1
    return result
