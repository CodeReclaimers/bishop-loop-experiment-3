def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    # Find end of number using rfind to avoid character-by-character scanning
    end_pos = pos
    while end_pos < n and text[end_pos] in _DIGIT_CHARS:
        end_pos += 1
    # Handle decimal point and exponent
    if end_pos < n and text[end_pos] == ".":
        end_pos += 1
        while end_pos < n and text[end_pos] in _DIGIT_CHARS:
            end_pos += 1
    if end_pos < n and text[end_pos] in ("e", "E"):
        end_pos += 1
        if end_pos < n and text[end_pos] in ("+", "-"):
            end_pos += 1
        while end_pos < n and text[end_pos] in _DIGIT_CHARS:
            end_pos += 1
    s = text[start:end_pos]
    if "." in s or "e" in s.lower():
        return float(s), end_pos
    return int(s), end_pos
