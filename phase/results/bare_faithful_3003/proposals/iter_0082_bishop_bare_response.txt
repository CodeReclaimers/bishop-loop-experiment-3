def _skip_ws(text: str, pos: int) -> int:
    if pos >= len(text):
        return pos
    # Find the first whitespace character using str.find with all whitespace bytes
    # This replaces the per-character loop with a single O(1) operation per run
    ws_bytes = b'\x20\x09\x0a\x0d'  # space, tab, newline, carriage return
    # Convert text to bytes for efficient searching
    text_bytes = text.encode('utf-8')
    # Find first non-whitespace byte
    start_pos = pos
    while start_pos < len(text_bytes):
        if text_bytes[start_pos] not in ws_bytes:
            break
        start_pos += 1
    return start_pos
