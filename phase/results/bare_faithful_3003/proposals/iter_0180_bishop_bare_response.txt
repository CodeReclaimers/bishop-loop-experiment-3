def _skip_ws(text: str, pos: int, non_ws_positions: list[int]) -> int:
    # Binary search for the next non-whitespace position
    if pos >= len(text):
        return pos
    left, right = 0, len(non_ws_positions) - 1
    while left <= right:
        mid = (left + right) // 2
        mid_pos = non_ws_positions[mid]
        if mid_pos == pos:
            return mid_pos + 1
        elif mid_pos < pos:
            left = mid + 1
        else:
            right = mid - 1
    # If we're here, pos is after the last non-whitespace char
    if left >= len(non_ws_positions):
        return len(text)

def _parse_object(text: str, pos: int) -> tuple[dict, int]:
    if text[pos] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos += 1
    out = {}
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "}":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != '"':
            raise JSONParseError(f"expected string key at {pos}")
        key, pos = _parse_string(text, pos)
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != ":":
            raise JSONParseError(f"expected ':' at {pos}")
        pos += 1
        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)

        pos = _skip_ws(text, pos)
        if pos >= len(text):

        if c == ",":
            pos += 1
