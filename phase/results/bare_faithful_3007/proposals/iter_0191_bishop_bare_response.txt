def _build_next_non_ws_table(text: str) -> list[int]:
    # Build a table where table[i] gives the index of the next non-whitespace
    # character starting from index i, or i if i is already non-whitespace.
    table = [0] * len(text)
    i = len(text) - 1
    while i >= 0:
        if text[i] in _WS_CHARS:
            if i == len(text) - 1:
                table[i] = i + 1
            else:
                table[i] = table[i + 1]
        else:
            table[i] = i
        i -= 1
    return table
