def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    is_float = False
    negative = False
    
    if pos < n and text[pos] == "-":
        negative = True
        pos = pos + 1
    
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    
    # Integer part
    integer_part = 0
    if text[pos] == "0":
        pos = pos + 1
    elif text[pos] in _DIGIT_CHARS:
        while pos < n and text[pos] in _DIGIT_CHARS:
            integer_part = integer_part * 10 + int(text[pos])
            pos = pos + 1
    
    # Fractional part
    fractional_part = 0
    fractional_digits = 0
    if pos < n and text[pos] == ".":
        is_float = True
        pos = pos + 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number fractional at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            fractional_part = fractional_part * 10 + int(text[pos])
            fractional_digits += 1
            pos = pos + 1
    
    # Exponent part
    exponent = 0
    if pos < n and text[pos] in ("e", "E"):
        is_float = True
        pos = pos + 1
        exp_negative = False
        if pos < n and text[pos] in ("+", "-"):
            exp_negative = text[pos] == "-"
            pos = pos + 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number exponent at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            exponent = exponent * 10 + int(text[pos])
            pos = pos + 1
        if exp_negative:
            exponent = -exponent
    
    # Build the final number
    if not is_float:
        result = integer_part
        if negative:
            result = -result
        return result, pos
    
    # For floats, build the number from parts
    result = integer_part
    if fractional_digits > 0:
        result += fractional_part / (10 ** fractional_digits)
    if exponent != 0:
        result *= (10 ** exponent)
    if negative:
        result = -result
    return result, pos
