def _skip_ws(text: str, pos: int) -> int:
    # Skip whitespace using find calls to jump directly to next whitespace char
    if not _WS_CHARS:
        return pos
    while pos < len(text):
        # Find the earliest occurrence of any whitespace character
        next_ws = len(text)
        for ch in _WS_CHARS:
            idx = text.find(ch, pos)
            if idx != -1 and idx < next_ws:
                next_ws = idx
        if next_ws == len(text):
            break
        pos = next_ws + 1
    return pos
