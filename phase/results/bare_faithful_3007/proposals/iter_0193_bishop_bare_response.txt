def _build_next_non_ws_table() -> list[int]:
    """Build a table for O(1) whitespace skipping."""
    table = [0] * 256
    for i in range(256):
        c = chr(i)
        if c in _WS_CHARS:
            table[i] = 1  # Mark as whitespace

            table[i] = 0  # Mark as non-whitespace
