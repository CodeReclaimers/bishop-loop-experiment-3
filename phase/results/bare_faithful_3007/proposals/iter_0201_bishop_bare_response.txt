def _build_ws_table(text: str) -> list[int]:
    """Build a table mapping each position to the next non-whitespace position."""
    n = len(text)
    ws_table = [0] * n
    pos = n
    while pos > 0:
        pos -= 1
        if text[pos] in _WS_CHARS:
            ws_table[pos] = ws_table[pos + 1]
        else:
            ws_table[pos] = pos + 1
    return ws_table


def _skip_ws(text: str, pos: int, ws_table: list[int]) -> int:
    # Skip whitespace using pre-computed table for O(1) lookups
    return ws_table[pos]
