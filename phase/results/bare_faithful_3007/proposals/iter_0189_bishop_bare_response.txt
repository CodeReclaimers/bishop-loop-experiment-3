def _build_next_non_ws_table() -> list[int]:
    # Build a table mapping each possible byte value to the next non-whitespace byte index
    table = [0] * 256
    for i in range(256):
        if chr(i) in _WS_CHARS:
            table[i] = 1  # Skip one character
        else:
            table[i] = 0  # No skip needed
    return table
