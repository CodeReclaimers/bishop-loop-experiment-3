import re

_NUMBER_RE = re.compile(r"""
    (?P<number>
        -?  # optional minus
        (?:0|[1-9][0-9]*)  # integer part
        (?:\.[0-9]+)?  # optional fractional part
        (?:[eE][+-]?[0-9]+)?  # optional exponent
    )
""", re.VERBOSE)

def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    match = _NUMBER_RE.match(text, pos)
    if not match:
        raise JSONParseError(f"bad number at {pos}")
    s = match.group("number")
    end_pos = pos + len(s)
    # Use float() for everything since it handles both int and float cases
    # and is more efficient than checking for decimal point
    num = float(s)
    # If it's actually an integer value, convert to int
    if num.is_integer():
        return int(num), end_pos
    return num, end_pos
