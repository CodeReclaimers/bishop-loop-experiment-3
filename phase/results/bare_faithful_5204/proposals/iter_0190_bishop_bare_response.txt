def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    # State machine for number parsing
    n = len(text)
    start = pos
    state = "sign"
    while pos < n:
        c = text[pos]
        if state == "sign":
            if c == "-":
                pos += 1
                state = "integer"
            else:
                state = "integer"
        elif state == "integer":
            if c == "0":
                pos += 1
                state = "fraction"
            elif c in _DIGIT_CHARS:
                pos += 1
            else:
                raise JSONParseError(f"bad number at {pos}")
                break
        elif state == "fraction":
            if c == ".":
                pos += 1
                state = "fraction_digits"
            elif c in ("e", "E"):
                pos += 1
                state = "exponent_sign"
            else:
                break
        elif state == "fraction_digits":
            if c in _DIGIT_CHARS:
                pos += 1
            else:
                state = "exponent"
        elif state == "exponent":
            if c in ("e", "E"):
                pos += 1
                state = "exponent_sign"
            else:
                break
        elif state == "exponent_sign":
            if c in ("+", "-"):
                pos += 1
                state = "exponent_digits"
            elif c in _DIGIT_CHARS:
                state = "exponent_digits"
            else:
                raise JSONParseError(f"bad number exponent at {pos}")
                break
        elif state == "exponent_digits":
            if c in _DIGIT_CHARS:
                pos += 1
            else:
                break
        else:
            break
    s = text[start:pos]
    # Determine if it's a float
    is_float = ("." in s) or ("e" in s.lower())
    if is_float:
        return float(s), pos
    return int(s), pos
