def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    # State machine for number parsing
    state = 0  # 0: integer part, 1: fractional part, 2: exponent part
    while pos < n:
        c = text[pos]
        if c in _DIGIT_CHARS:
            pos += 1
        elif c == "." and state == 0:
            state = 1
            pos += 1
        elif (c == "e" or c == "E") and (state == 0 or state == 1):
            state = 2
            pos += 1
            if pos < n and text[pos] in ("+", "-"):
                pos += 1
        elif c in ("+", "-") and state == 2:
            pos += 1
        else:
            break
    if pos == start:
        raise JSONParseError(f"bad number at {pos}")
    s = text[start:pos]
    # Determine if it's a float or int
    is_float = state != 0
    if is_float:
        return float(s), pos
    return int(s), pos
