def _skip_ws(text: str, pos: int) -> int:
    # Use a fixed-size buffer to skip whitespace characters efficiently
    buffer_size = 64
    while pos < len(text):
        # Fill buffer with whitespace characters
        buffer_pos = 0
        while buffer_pos < buffer_size and pos + buffer_pos < len(text) and text[pos + buffer_pos] in _WS_CHARS:
            buffer_pos += 1
        # Skip the buffer of whitespace
        pos += buffer_pos
        # Check if we've reached a non-whitespace character
        if pos >= len(text) or text[pos] not in _WS_CHARS:
            break
    return pos
