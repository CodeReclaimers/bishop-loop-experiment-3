def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    # State machine for number parsing
    state = "start"
    while pos < n:
        c = text[pos]
        if state == "start":
            if c == "-":
                pos += 1
                state = "after_neg"
            elif c == "0":
                pos += 1
                state = "after_zero"
            elif c in _DIGIT_CHARS:
                pos += 1
                state = "digits"
            else:
                raise JSONParseError(f"bad number at {pos}")
        elif state == "after_neg":
            if c == "0":
                pos += 1
                state = "after_zero"
            elif c in _DIGIT_CHARS:
                pos += 1
                state = "digits"
            else:
                raise JSONParseError(f"bad number at {pos}")
        elif state == "after_zero":
            if c == ".":
                pos += 1
                state = "after_dot"
            elif c in ("e", "E"):
                pos += 1
                state = "after_exp"
            else:
                break
        elif state == "digits":
            if c in _DIGIT_CHARS:
                pos += 1

                pos += 1
                state = "after_dot"
            elif c in ("e", "E"):
                pos += 1
                state = "after_exp"
            else:
                break

            if c in _DIGIT_CHARS:
                pos += 1
                state = "frac_digits"

            if c in _DIGIT_CHARS:
                pos += 1

                pos += 1

                raise JSONParseError(f"bad number exponent at {pos}")

            if c in _DIGIT_CHARS:
                pos += 1
