def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    # Find the end of the number
    end = pos
    while end < len(text) and (text[end] in "0123456789+-eE."):
        end += 1
    s = text[pos:end]
    if not s:
        raise JSONParseError(f"bad number at {pos}")
    try:
        if "." in s or "e" in s.lower():
            return float(s), end
        return int(s), end

        raise JSONParseError(f"bad number at {pos}")
