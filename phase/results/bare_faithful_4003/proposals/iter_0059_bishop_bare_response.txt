def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    # Use iterative loop with enumerate to parse number
    start = pos
    for i, c in enumerate(text[pos:], pos):
        if c not in _DIGIT_CHARS and c not in ".eE+-":
            break
    else:
        i = len(text)
    s = text[start:i]
    if not s:
        raise JSONParseError(f"bad number at {pos}")
    pos = i
    if "." in s or "e" in s.lower():
        return float(s), pos
    return int(s), pos
