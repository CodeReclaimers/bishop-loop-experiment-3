def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    start = pos
    # Handle optional leading minus
    if text[pos] == "-":
        pos += 1
    # Handle integer part
    if pos >= len(text):
        raise JSONParseError(f"bad number at {start}")
    if text[pos] == "0":
        pos += 1
    else:
        if text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number at {start}")
        pos += 1
        while pos < len(text) and text[pos] in _DIGIT_CHARS:
            pos += 1
    # Handle fractional part
    if pos < len(text) and text[pos] == ".":
        pos += 1
