def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos = pos + 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    
    # State machine for number parsing
    state = 0  # 0: integer part, 1: decimal point, 2: fractional part, 3: exponent, 4: exponent sign, 5: exponent digits
    while pos < n:
        c = text[pos]
        if state == 0:  # integer part
            if c == "0":
                pos = pos + 1
            elif c in _DIGIT_CHARS:
                pos = pos + 1
                while pos < n and text[pos] in _DIGIT_CHARS:
                    pos = pos + 1
                state = 1
            else:
                raise JSONParseError(f"bad number at {pos}")
        elif state == 1:  # decimal point
            if c == ".":
                state = 2
                pos = pos + 1
                if pos >= n or text[pos] not in _DIGIT_CHARS:
                    raise JSONParseError(f"bad number fractional at {pos}")
                while pos < n and text[pos] in _DIGIT_CHARS:
                    pos = pos + 1
            elif c in ("e", "E"):
                state = 3
                pos = pos + 1
                if pos < n and text[pos] in ("+", "-"):
                    pos = pos + 1
                    state = 4
                if pos >= n or text[pos] not in _DIGIT_CHARS:
                    raise JSONParseError(f"bad number exponent at {pos}")
                while pos < n and text[pos] in _DIGIT_CHARS:
                    pos = pos + 1
            else:
                break
        elif state == 2:  # fractional part
            if c in ("e", "E"):
                state = 3
                pos = pos + 1
                if pos < n and text[pos] in ("+", "-"):
                    pos = pos + 1
                    state = 4
                if pos >= n or text[pos] not in _DIGIT_CHARS:
                    raise JSONParseError(f"bad number exponent at {pos}")
                while pos < n and text[pos] in _DIGIT_CHARS:
                    pos = pos + 1
            else:
                break
        elif state == 3:  # exponent
            if c in _DIGIT_CHARS:
                pos = pos + 1
                while pos < n and text[pos] in _DIGIT_CHARS:
                    pos = pos + 1
            else:
                break
        elif state == 4:  # exponent sign
            if c in _DIGIT_CHARS:
                pos = pos + 1
                while pos < n and text[pos] in _DIGIT_CHARS:
                    pos = pos + 1
                state = 3
            else:
                raise JSONParseError(f"bad number exponent at {pos}")
    
    s = text[start:pos]
    # Check if it's a float by looking for decimal point or exponent
    is_float = "." in s or "e" in s.lower()
    if is_float:
        return float(s), pos
    return int(s), pos
