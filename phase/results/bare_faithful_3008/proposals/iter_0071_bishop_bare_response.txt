# Precompile the regex for skipping whitespace
_SKIP_WS_RE = re.compile(r"^\s*")

    if pos >= len(text):

        raise JSONParseError(f"trailing content at position {pos}")
    return value


def _skip_ws(text: str, pos: int) -> int:
    # Use a single regex match to skip all whitespace at once
    match = _SKIP_WS_RE.match(text, pos)
