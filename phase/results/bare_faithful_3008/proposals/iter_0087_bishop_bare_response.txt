def _skip_ws(text: str, pos: int) -> int:
    # Use str.find for each whitespace character with lookup table
    while pos < len(text):
        next_ws = min(
            text.find(c, pos) if c != '\n' else pos + 1
            for c in _WS_CHARS
            if text.find(c, pos) != -1
        )
        if next_ws == pos:
            pos += 1
        else:
            return pos
    return pos
