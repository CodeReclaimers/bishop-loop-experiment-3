def _skip_ws(text: str, pos: int) -> int:
    # Use text.find() calls to locate the next non-whitespace character
    while pos < len(text):
        # Find the next occurrence of any whitespace character
        next_tab = text.find('\t', pos)
        next_nl = text.find('\n', pos)
        next_cr = text.find('\r', pos)
        next_sp = text.find(' ', pos)
        
        # Find the minimum of all found positions
        min_pos = len(text)
        if next_tab != -1:
            min_pos = min(min_pos, next_tab)
        if next_nl != -1:
            min_pos = min(min_pos, next_nl)
        if next_cr != -1:
            min_pos = min(min_pos, next_cr)
        if next_sp != -1:
            min_pos = min(min_pos, next_sp)
        
        # If no whitespace found, we're done
        if min_pos == len(text):
            break
            
        # If we found a whitespace character, advance past it
        if min_pos == pos:
            pos += 1
        else:
            # Found a non-whitespace character at min_pos
            break
