def _is_hex4(s: str) -> bool:
    if len(s) != 4:
        return False
    # Create translation table to map valid hex chars to a marker
    # and invalid chars to None, then check if all chars are valid
    trans_table = str.maketrans({ch: None for ch in s if ch not in _HEX_CHARS})
    return not bool(trans_table)
