    # Use str.find calls to skip whitespace in a single operation
    while True:
        next_space = text.find(" ", pos)
        next_tab = text.find("\t", pos)
        next_newline = text.find("\n", pos)
        next_cr = text.find("\r", pos)
        
        # Find the minimum position of any whitespace character
        min_pos = len(text)
        if next_space != -1 and next_space < min_pos:
            min_pos = next_space
        if next_tab != -1 and next_tab < min_pos:
            min_pos = next_tab
        if next_newline != -1 and next_newline < min_pos:
            min_pos = next_newline
        if next_cr != -1 and next_cr < min_pos:
            min_pos = next_cr
            
        # If no whitespace found, return current position
        if min_pos == len(text):
            return pos
            
        # If we found whitespace at current position, advance and continue
        if min_pos == pos:
            pos += 1
        else:
            # Found whitespace at a later position, return that position
            return min_pos
