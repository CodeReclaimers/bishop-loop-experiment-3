def _skip_ws(text: str, pos: int) -> int:
    # Use str.find calls to locate next whitespace character in one pass
    while pos < len(text):
        next_tab = text.find('\t', pos)
        next_nl = text.find('\n', pos)
        next_cr = text.find('\r', pos)
        next_sp = text.find(' ', pos)
        next_ws = min(
            x for x in (next_tab, next_nl, next_cr, next_sp) if x != -1
        ) if any(x != -1 for x in (next_tab, next_nl, next_cr, next_sp)) else -1
        if next_ws == -1:
            break
        pos = next_ws + 1
    return pos
