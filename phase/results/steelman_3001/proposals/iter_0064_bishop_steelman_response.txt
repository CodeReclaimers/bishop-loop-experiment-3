import re

# Precompile the number regex for performance
_NUMBER_RE = re.compile(r'^-?(?:0|[1-9]\d*)(?:\.\d+)?(?:[eE][+-]?\d+)?')

def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    # Match the full number token using regex
    match = _NUMBER_RE.match(text, pos)
    if not match:
        raise JSONParseError(f"bad number at {pos}")
    s = match.group(0)
    new_pos = pos + len(s)
    # Determine if it's a float or int by checking for decimal or exponent
    is_float = '.' in s or 'e' in s.lower()
    if is_float:
        return float(s), new_pos
    return int(s), new_pos
