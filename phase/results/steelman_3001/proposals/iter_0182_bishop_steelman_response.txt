_NUMBER_RE = re.compile(r"""
    -?              # optional minus
    (?:0|[1-9]\d*)  # integer part: 0 or digits without leading zero
    (?:\.\d+)?      # optional fractional part
    (?:[eE][+-]?\d+)? # optional exponent
""", re.VERBOSE)
