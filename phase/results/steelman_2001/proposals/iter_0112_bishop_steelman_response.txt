import re

# Precompile the number regex for performance
_NUMBER_RE = re.compile(r"""
    (?P<number>
        -?                          # optional minus
        (?:0|[1-9]\d*)              # integer part: 0 or digits without leading zero
        (?:\.\d+)?                  # optional fractional part
        (?:[eE][+-]?\d+)?           # optional exponent part
    )
""", re.VERBOSE)

def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    match = _NUMBER_RE.match(text, pos)
    if not match:
        raise JSONParseError(f"bad number at {pos}")
    s = match.group("number")
    new_pos = pos + len(s)
    # Check if it's a float by presence of decimal or exponent
    if "." in s or "e" in s.lower():
        return float(s), new_pos
    return int(s), new_pos
