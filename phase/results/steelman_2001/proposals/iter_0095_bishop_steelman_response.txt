import re

# Precompile regex for number parsing
_NUMBER_RE = re.compile(r"""
    (?P<number>
        -?                             # optional minus
        (?:0|[1-9]\d*)                 # integer part (no leading zeros except for "0")
        (?:\.\d+)?                     # optional fractional part
        (?:[eE][+-]?\d+)?              # optional exponent
    )
""", re.VERBOSE)

def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    # Match the full number literal using regex
    match = _NUMBER_RE.match(text, pos)
    if not match:
        raise JSONParseError(f"bad number at {pos}")
    
    s = match.group('number')
    end_pos = pos + len(s)
    
    # Validate that the matched number conforms to JSON rules
    # Check for invalid leading zeros (e.g., "01", "001")
    if len(s) > 1 and s[0] == '0' and s[1] != '.':
        # This check is necessary for cases like "01", "001", etc.
        # but not for "0", "0.5", etc.
        if not (s.startswith('0.') or s.startswith('0e') or s.startswith('0E')):
            raise JSONParseError(f"bad number at {pos}")
    
    # Convert to appropriate type
    if '.' in s or 'e' in s.lower():
        return float(s), end_pos
    return int(s), end_pos
