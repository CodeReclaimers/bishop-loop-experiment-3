Additionally, `str.find` doesn't handle the case where the input is all whitespace, and the resulting code would be less readable and harder to maintain.

def _skip_ws(text: str, pos: int) -> int:
    # Use str.translate to efficiently skip whitespace
    if pos >= len(text):
        return pos
    # Create translation table to map whitespace to a sentinel character
    whitespace_table = str.maketrans(' \t\n\r', '    ')

    translated = text[pos:].translate(whitespace_table)
