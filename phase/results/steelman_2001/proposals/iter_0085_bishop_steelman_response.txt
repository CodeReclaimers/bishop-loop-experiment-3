import re

# Precompile regex for JSON numbers
_JSON_NUMBER_RE = re.compile(r"""
    -? (?: 0 | [1-9]\d* ) (?: \. \d+ )? (?: [eE] [+\-]? \d+ )?
""", re.VERBOSE)

def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    # Find the longest valid JSON number starting at pos
    match = _JSON_NUMBER_RE.match(text, pos)
    if not match:
        raise JSONParseError(f"bad number at {pos}")
    s = match.group(0)
    pos += len(s)
    # Use Python's built-in conversion which handles all edge cases

        if '.' in s or 'e' in s.lower():

            return int(s), pos
