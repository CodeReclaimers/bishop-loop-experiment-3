To make the JSON parser faster, we can optimize string concatenation in `_parse_string` by using a `StringBuilder` class instead of repeatedly calling `str.join()`. This approach reduces overhead from multiple string creations and improves performance significantly.

```python
class StringBuilder:
    def __init__(self):
        self._buffer = []

    def append(self, char):
        self._buffer.append(char)

    def get_string(self):
        return ''.join(self._buffer)


def _parse_string(text: str, pos: int) -> tuple[str, int]:
    if text[pos] != '"':
        raise JSONParseError(f"expected string at {pos}")
    pos = pos + 1
    builder = StringBuilder()
    while pos < len(text):
        c = text[pos]
        if c == '"':
            return builder.get_string(), pos + 1
        if c == "\\":
            pos = pos + 1
            if pos >= len(text):
                raise JSONParseError("unterminated string escape")
            esc = text[pos]
            if esc in _ESCAPE_CHARS:
                builder.append(_ESCAPE_CHARS[esc])
            elif esc == "u":
                if pos + 4 >= len(text):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                hex4 = text[pos + 1 : pos + 5]
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if pos + 5 >= len(text) or text[pos + 5] != "\\" or pos + 6 >= len(text) or text[pos + 6] != "u":
                        raise JSONParseError(f"unpaired high surrogate at {pos}")
                    if pos + 10 >= len(text):
                        raise JSONParseError(f"bad low surrogate at {pos}")
                    hex4b = text[pos + 7 : pos + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    builder.append(chr(cp))
                    pos = pos + 11
                    continue
                builder.append(chr(cp))
                pos = pos + 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {pos}")
            pos = pos + 1
            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {pos}")
        builder.append(c)
        pos = pos + 1
    raise JSONParseError("unterminated string")
```

This change uses a `StringBuilder` to efficiently concatenate characters, reducing the number of string creations and improving the overall performance of the parsing process.
