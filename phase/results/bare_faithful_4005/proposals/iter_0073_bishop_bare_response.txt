def _parse_object(text: str, pos: int) -> tuple[dict, int]:
    if text[pos] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos = pos + 1
    out: dict = {}
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "}":
        return out, pos + 1
    
    # Stack-based parsing for object
    stack = [(out, False)]  # (current_dict, expecting_value)
    while stack:
        current_dict, expecting_value = stack[-1]
        if expecting_value:
            # Parse value
            pos = _skip_ws(text, pos)
            if pos >= len(text):
                raise JSONParseError("unexpected end of input")
            value, pos = _parse_value(text, pos)
            current_dict[stack[-1][0]] = value
            stack[-1] = (current_dict, False)
        else:
            # Parse key or end
            pos = _skip_ws(text, pos)

            stack[-1] = (current_dict, True)
