def _parse_value(text: str, pos: int) -> tuple[Any, int]:
    # Check cache for previously parsed value
    cache_key = (text, pos)
    if cache_key in _parse_cache:
        return _parse_cache[cache_key]
    
    pos = _skip_ws(text, pos)
    if pos >= len(text):
        raise JSONParseError("unexpected end of input")
    c = text[pos]
    if c == "{":
        result = _parse_object(text, pos)
    if c == "[":
        result = _parse_array(text, pos)
    if c == '"':
        result = _parse_string(text, pos)
    if c == "t" or c == "f":
        result = _parse_bool(text, pos)
    if c == "n":
        result = _parse_null(text, pos)
    if c == "-" or c in _DIGIT_CHARS:
