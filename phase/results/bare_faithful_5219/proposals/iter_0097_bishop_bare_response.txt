def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    if text[pos] == "0":
        pos += 1
    elif text[pos] in _DIGIT_CHARS:
        # Use running sum for digit scanning
        value = 0
        while pos < n and text[pos] in _DIGIT_CHARS:
            value = value * 10 + (ord(text[pos]) - ord('0'))
            pos += 1
        # Convert back to string for float/int conversion
        s = str(value)
        # Handle fractional part
        is_float = False
        if pos < n and text[pos] == ".":
            is_float = True
            pos += 1
            if pos >= n or text[pos] not in _DIGIT_CHARS:
                raise JSONParseError(f"bad number fractional at {pos}")
            # Use running sum for fractional part
            frac_value = 0
            frac_digits = 0
            while pos < n and text[pos] in _DIGIT_CHARS:
                frac_value = frac_value * 10 + (ord(text[pos]) - ord('0'))
                frac_digits += 1
                pos += 1
            s += "." + str(frac_value)
            # Adjust for decimal places
            s = s[:-frac_digits] + str(frac_value)
        # Handle exponent
        if pos < n and text[pos] in ("e", "E"):
            is_float = True
            pos += 1
            if pos < n and text[pos] in ("+", "-"):
                pos += 1
            if pos >= n or text[pos] not in _DIGIT_CHARS:
                raise JSONParseError(f"bad number exponent at {pos}")
            # Use running sum for exponent
            exp_value = 0
            while pos < n and text[pos] in _DIGIT_CHARS:
                exp_value = exp_value * 10 + (ord(text[pos]) - ord('0'))
                pos += 1
            s += "e" + str(exp_value)
    else:
        raise JSONParseError(f"bad number at {pos}")
    if is_float:
        return float(s), pos
    return int(s), pos
