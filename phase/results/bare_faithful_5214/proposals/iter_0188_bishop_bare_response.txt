def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    # Use regex to match the number pattern
    match = re.match(r'0|([1-9][0-9]*)(\.[0-9]+)?([eE][+-]?[0-9]+)?', text[pos:])
    if not match:
        raise JSONParseError(f"bad number at {pos}")
    matched = match.group(0)
    end = pos + len(matched)
    s = text[start:end]
    # Check if it's a float by presence of decimal point or exponent
    is_float = '.' in matched or 'e' in matched.lower()
    if is_float:
        return float(s), end
    return int(s), end
