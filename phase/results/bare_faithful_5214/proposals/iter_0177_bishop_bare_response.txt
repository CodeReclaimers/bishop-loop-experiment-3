def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    
    # State machine for number parsing
    state = "integer"
    while pos < n:
        c = text[pos]
        if state == "integer":
            if c == "0":
                pos += 1
            elif c in _DIGIT_CHARS:
                pos += 1
            else:
                raise JSONParseError(f"bad number at {pos}")
                break
            state = "fractional" if c == "." else "exponent" if c in ("e", "E") else "integer"
        elif state == "fractional":
            if c == ".":
                pos += 1
                state = "fraction_digits"
            elif c in ("e", "E"):
                pos += 1
                state = "exponent_sign"
            else:
                break

            if c in _DIGIT_CHARS:
                pos += 1
            else:
                state = "exponent" if c in ("e", "E") else "done"

            if c in ("e", "E"):
                pos += 1
                state = "exponent_sign"

            state = "exponent_digits"

            if c in _DIGIT_CHARS:
