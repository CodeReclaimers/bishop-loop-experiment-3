"""Deliberately slow but correct JSON parser. Editable surface.

This is the starting point for the optimization loop. It is correct against
`json.loads` on the full corpus but is intentionally written in a way that
leaves significant performance on the table:

- Whitespace is skipped via a per-call Python loop instead of a regex.
- Numbers are scanned character-by-character with explicit predicates.
- Object/array building uses string concatenation and rebuilt-on-each-step
  collection (`out = out + [v]`, `out = dict(out); out[k] = v`).
- Recursive descent with redundant whitespace handling at every level.

The optimization target is wall-clock time to parse the 200-case fixed
corpus. The candidate must keep `parse(text) -> Any` as the public entry
point and must raise `JSONParseError` for malformed inputs.
"""
from __future__ import annotations

import re
from typing import Any


class JSONParseError(ValueError):
    """Raised when the input is not valid JSON."""


# Deliberately use lists not sets — every membership check is O(k) instead of O(1).
_WS_CHARS = frozenset(" \t\n\r")
_DIGIT_CHARS = frozenset("0123456789")


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    pos = _skip_ws(text, 0)
    if pos >= len(text):
        raise JSONParseError("empty input")
    value, pos = _parse_value(text, pos)
    pos = _skip_ws(text, pos)
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value


def _skip_ws(text: str, pos: int) -> int:
    n = len(text)
    while pos < n and text[pos] in _WS_CHARS:
        pos += 1
    return pos


def _parse_value(text: str, pos: int) -> tuple[Any, int]:
    pos = _skip_ws(text, pos)
    if pos >= len(text):
        raise JSONParseError("unexpected end of input")
    c = text[pos]
    if c == "{":
        return _parse_object(text, pos)
    if c == "[":
        return _parse_array(text, pos)
    if c == '"':
        return _parse_string(text, pos)
    if c == "t" or c == "f":
        return _parse_bool(text, pos)
    if c == "n":
        return _parse_null(text, pos)
    if c == "-" or c in _DIGIT_CHARS:
        return _parse_number(text, pos)
    raise JSONParseError(f"unexpected character {c!r} at position {pos}")


def _parse_object(text: str, pos: int) -> tuple[dict, int]:
    if text[pos] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos += 1
    out: dict = {}
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "}":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != '"':
            raise JSONParseError(f"expected string key at {pos}")
        key, pos = _parse_string(text, pos)
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != ":":
            raise JSONParseError(f"expected ':' at {pos}")
        pos += 1
        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)
        out[key] = value
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated object")
        c = text[pos]
        if c == ",":
            pos += 1
            continue
        if c == "}":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or '}}' at {pos}")


def _parse_array(text: str, pos: int) -> tuple[list, int]:
    if text[pos] != "[":
        raise JSONParseError(f"expected '[' at {pos}")
    pos += 1
    out: list = []
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "]":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)
        out.append(value)
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated array")
        c = text[pos]
        if c == ",":
            pos += 1
            continue
        if c == "]":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or ']' at {pos}")


def _parse_string(text: str, pos: int) -> tuple[str, int]:
    if text[pos] != '"':
        raise JSONParseError(f"expected '\"' at {pos}")
    pos = pos + 1
    n = len(text)
    out = ""
    while pos < n:
        c = text[pos]
        if c == '"':
            return out, pos + 1
        if c == "\\":
            pos = pos + 1
            if pos >= n:
                raise JSONParseError("unterminated string escape")
            esc = text[pos]
            if esc == '"':
                out = out + '"'
            elif esc == "\\":
                out = out + "\\"
            elif esc == "/":
                out = out + "/"
            elif esc == "b":
                out = out + "\b"
            elif esc == "f":
                out = out + "\f"
            elif esc == "n":
                out = out + "\n"
            elif esc == "r":
                out = out + "\r"
            elif esc == "t":
                out = out + "\t"
            elif esc == "u":
                if pos + 4 >= n:
                    raise JSONParseError(f"bad unicode escape at {pos}")
                hex4 = text[pos + 1 : pos + 5]
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if pos + 5 >= n or text[pos + 5] != "\\" or pos + 6 >= n or text[pos + 6] != "u":
                        raise JSONParseError(f"unpaired high surrogate at {pos}")
                    if pos + 10 >= n:
                        raise JSONParseError(f"bad low surrogate at {pos}")
                    hex4b = text[pos + 7 : pos + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    out = out + chr(cp)
                    pos = pos + 11
                    continue
                out = out + chr(cp)
                pos = pos + 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {pos}")
            pos = pos + 1
            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {pos}")
        out = out + c
        pos = pos + 1
    raise JSONParseError("unterminated string")


def _is_hex4(s: str) -> bool:
    if len(s) != 4:
        return False
    for ch in s:
        if not (("0" <= ch <= "9") or ("a" <= ch <= "f") or ("A" <= ch <= "F")):
            return False
    return True


def _parse_bool(text: str, pos: int) -> tuple[bool, int]:
    if text[pos : pos + 4] == "true":
        return True, pos + 4
    if text[pos : pos + 5] == "false":
        return False, pos + 5
    raise JSONParseError(f"bad bool at {pos}")


def _parse_null(text: str, pos: int) -> tuple[None, int]:
    if text[pos : pos + 4] == "null":
        return None, pos + 4
    raise JSONParseError(f"bad null at {pos}")


def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    if text[pos] == "0":
        pos += 1
    elif text[pos] in _DIGIT_CHARS:
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    else:
        raise JSONParseError(f"bad number at {pos}")
    is_float = False
    if pos < n and text[pos] == ".":
        is_float = True
        pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number fractional at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    if pos < n and text[pos] in ("e", "E"):
        is_float = True
        pos += 1
        if pos < n and text[pos] in ("+", "-"):
            pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number exponent at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    s = text[start:pos]
    if is_float:
        return float(s), pos
    return int(s), pos

This is the starting point for the optimization loop. It is correct against

- Numbers are scanned character-by-character with explicit predicates.
- Object/array building uses string concatenation and rebuilt-on-each-step
  collection (`out = out + [v]`, `out = dict(out); out[k] = v`).
- Recursive descent with redundant whitespace handling at every level.

The optimization target is wall-clock time to parse the 200-case fixed
corpus. The candidate must keep `parse(text) -> Any` as the public entry
point and must raise `JSONParseError` for malformed inputs.
"""
from __future__ import annotations

import re
from typing import Any
import io


class JSONParseError(ValueError):
    """Raised when the input is not valid JSON."""


# Deliberately use lists not sets — every membership check is O(k) instead of O(1).
_WS_CHARS = frozenset(" \t\n\r")
_DIGIT_CHARS = frozenset("0123456789")


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    stream = io.StringIO(text)
    pos = _skip_ws_stream(stream)
    if pos >= len(text):
        raise JSONParseError("empty input")
    value, pos = _parse_value_stream(stream, pos)
    pos = _skip_ws_stream(stream)
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value


def _skip_ws_stream(stream: io.StringIO) -> int:
    pos = stream.tell()
    while True:
        char = stream.read(1)
        if not char or char not in _WS_CHARS:
            stream.seek(pos)
            break
        pos = stream.tell()
    return pos


def _parse_value_stream(stream: io.StringIO, pos: int) -> tuple[Any, int]:
    stream.seek(pos)
    pos = _skip_ws_stream(stream)
    if pos >= len(stream.getvalue()):
        raise JSONParseError("unexpected end of input")
    c = stream.read(1)
    if c == "{":
        return _parse_object_stream(stream, pos)
    if c == "[":
        return _parse_array_stream(stream, pos)
    if c == '"':
        return _parse_string_stream(stream, pos)
    if c == "t" or c == "f":
        return _parse_bool_stream(stream, pos)
    if c == "n":
        return _parse_null_stream(stream, pos)
    if c == "-" or c in _DIGIT_CHARS:
        return _parse_number_stream(stream, pos)
    raise JSONParseError(f"unexpected character {c!r} at position {pos}")


def _parse_object_stream(stream: io.StringIO, pos: int) -> tuple[dict, int]:
    stream.seek(pos)
    if stream.read(1) != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    out: dict = {}
    pos = _skip_ws_stream(stream)
    if pos >= len(stream.getvalue()):
        raise JSONParseError("unexpected end of input")
    stream.seek(pos)
    if stream.read(1) == "}":
        return out, pos + 1
    while True:
        pos = _skip_ws_stream(stream)
        if pos >= len(stream.getvalue()):
            raise JSONParseError("unexpected end of input")
        stream.seek(pos)
        if stream.read(1) != '"':
            raise JSONParseError(f"expected string key at {pos}")
        key, pos = _parse_string_stream(stream, pos)
        pos = _skip_ws_stream(stream)
        if pos >= len(stream.getvalue()):
            raise JSONParseError("unexpected end of input")
        stream.seek(pos)
        if stream.read(1) != ":":
            raise JSONParseError(f"expected ':' at {pos}")
        pos += 1
        pos = _skip_ws_stream(stream)
        if pos >= len(stream.getvalue()):
            raise JSONParseError("unexpected end of input")
        value, pos = _parse_value_stream(stream, pos)
        out[key] = value
        pos = _skip_ws_stream(stream)
        if pos >= len(stream.getvalue()):
            raise JSONParseError("unterminated object")
        stream.seek(pos)
        c = stream.read(1)
        if c == ",":
            pos += 1
            continue
        if c == "}":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or '}}' at {pos}")


def _parse_array_stream(stream: io.StringIO, pos: int) -> tuple[list, int]:
    stream.seek(pos)
    if stream.read(1) != "[":
        raise JSONParseError(f"expected '[' at {pos}")
    out: list = []
    pos = _skip_ws_stream(stream)
    if pos >= len(stream.getvalue()):
        raise JSONParseError("unexpected end of input")
    stream.seek(pos)
    if stream.read(1) == "]":
        return out, pos + 1
    while True:
        pos = _skip_ws_stream(stream)
        if pos >= len(stream.getvalue()):
            raise JSONParseError("unexpected end of input")
        value, pos = _parse_value_stream(stream, pos)
        out.append(value)
        pos = _skip_ws_stream(stream)
        if pos >= len(stream.getvalue()):
            raise JSONParseError("unterminated array")
        stream.seek(pos)
        c = stream.read(1)
        if c == ",":
            pos += 1
            continue
        if c == "]":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or ']' at {pos}")


def _parse_string_stream(stream: io.StringIO, pos: int) -> tuple[str, int]:
    stream.seek(pos)
    if stream.read(1) != '"':
        raise JSONParseError(f"expected '\"' at {pos}")
    pos = pos + 1
    out = ""
    while True:
        stream.seek(pos)
        c = stream.read(1)
        if not c:
            raise JSONParseError("unterminated string")
        if c == '"':
            return out, pos + 1
        if c == "\\":
            pos = pos + 1
            stream.seek(pos)
            if not stream.read(1):
                raise JSONParseError("unterminated string escape")
            esc = stream.read(1)
            if esc == '"':
                out = out + '"'
            elif esc == "\\":
                out = out + "\\"
            elif esc == "/":
                out = out + "/"
            elif esc == "b":
                out = out + "\b"
            elif esc == "f":
                out = out + "\f"
            elif esc == "n":
                out = out + "\n"
            elif esc == "r":
                out = out + "\r"
            elif esc == "t":
                out = out + "\t"
            elif esc == "u":
                stream.seek(pos + 1)
                if pos + 4 >= len(stream.getvalue()):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                hex4 = stream.read(4)
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    stream.seek(pos + 5)
                    if pos + 5 >= len(stream.getvalue()) or stream.read(1) != "\\" or pos + 6 >= len(stream.getvalue()) or stream.read(1) != "u":
                        raise JSONParseError(f"unpaired high surrogate at {pos}")
                    if pos + 10 >= len(stream.getvalue()):
                        raise JSONParseError(f"bad low surrogate at {pos}")
                    stream.seek(pos + 7)
                    hex4b = stream.read(4)
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    out = out + chr(cp)
                    pos = pos + 11
                    continue
                out = out + chr(cp)
                pos = pos + 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {pos}")
            pos = pos + 1
            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {pos}")
        out = out + c
        pos = pos + 1
    raise JSONParseError("unterminated string")


def _is_hex4(s: str) -> bool:
    if len(s) != 4:
        return False
    for ch in s:
        if not (("0" <= ch <= "9") or ("a" <= ch <= "f") or ("A" <= ch <= "F")):
            return False
    return True


def _parse_bool_stream(stream: io.StringIO, pos: int) -> tuple[bool, int]:
    stream.seek(pos)
    s = stream.read(4)
    if s == "true":
        return True, pos + 4
    stream.seek(pos)
    s = stream.read(5)
    if s == "false":
        return False, pos + 5
    raise JSONParseError(f"bad bool at {pos}")


def _parse_null_stream(stream: io.StringIO, pos: int) -> tuple[None, int]:
    stream.seek(pos)
    s = stream.read(4)
    if s == "null":
        return None, pos + 4
    raise JSONParseError(f"bad null at {pos}")

    start = pos
    if stream.read(1) == "-":
        pos += 1
    if pos >= len(stream.getvalue()):
        raise JSONParseError(f"bad number at {pos}")
    stream.seek(pos)
    if stream.read(1) == "0":
        pos += 1
    elif stream.read(1) in _DIGIT_CHARS:
        while True:
            stream.seek(pos)
            if pos >= len(stream.getvalue()) or stream.read(1) not in _DIGIT_CHARS:
                break
            pos += 1
    else:
        raise JSONParseError(f"bad number at {pos}")
    is_float = False
    stream.seek(pos)
    if stream.read(1) == ".":
        is_float = True
        pos += 1
        stream.seek(pos)
        if pos >= len(stream.getvalue()) or stream.read(1) not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number fractional at {pos}")
        while True:

            if pos >= len(stream.getvalue()) or stream.read(1) not in _DIGIT_CHARS:
                break
            pos += 1
    if pos < len(stream.getvalue()):
        stream.seek(pos)
        if stream.read(1) in ("e", "E"):

            pos += 1

    stream.seek(start)
