        raise JSONParseError("input must be a string")
    pos = _skip_ws(text, 0)
    if pos >= len(text):
        raise JSONParseError("empty input")
    
    # Stack-based approach to avoid recursion
    stack = []
    current = None
    while pos < len(text):
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unexpected end of input")
        
        c = text[pos]
        if c == '{':
            stack.append(('object', {}, pos + 1))
            pos = _skip_ws(text, pos + 1)
            if pos < len(text) and text[pos] == '}':
                # Empty object
                pos += 1
                if stack:
                    stack.pop()
                else:
                    break
            continue
        elif c == '[':
            stack.append(('array', [], pos + 1))
            pos = _skip_ws(text, pos + 1)
            if pos < len(text) and text[pos] == ']':
                # Empty array
                pos += 1
                if stack:
                    stack.pop()
                else:
                    break
            continue
        elif c == '"':
            value, pos = _parse_string(text, pos)
            if stack:
                if stack[-1][0] == 'object':
                    # We're parsing an object key or value
                    if 'key' not in stack[-1][1]:
                        stack[-1][1]['key'] = value
                    else:
                        # This is a value
                        stack[-1][1]['value'] = value
                        # Add to object
                        if 'key' in stack[-1][1] and 'value' in stack[-1][1]:
                            stack[-1][1][0][stack[-1][1]['key']] = stack[-1][1]['value']
                            stack[-1][1]['key'] = None
                            stack[-1][1]['value'] = None

                stack.pop()
