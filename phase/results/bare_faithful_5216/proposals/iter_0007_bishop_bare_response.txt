def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    # Use regex to scan for numbers
    number_match = re.match(r'[-+]?(?:\d+\.?\d*|\.\d+)(?:[eE][+-]?\d+)?', text[pos:])
    if not number_match:
        raise JSONParseError(f"bad number at {pos}")
    number_str = number_match.group(0)
    pos += len(number_str)
    if '.' in number_str or 'e' in number_str.lower():
        return float(number_str), pos
    return int(number_str), pos
