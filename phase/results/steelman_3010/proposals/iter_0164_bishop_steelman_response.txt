def _skip_ws(text: str, pos: int) -> int:
    # Use str.translate to efficiently skip whitespace
    # Create translation table to replace all whitespace with a delimiter
    whitespace_trans = str.maketrans(' \t\n\r', '    ')
    # Translate the text from pos onward
    translated = text[pos:].translate(whitespace_trans)
