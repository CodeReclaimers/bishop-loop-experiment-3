def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    is_float = False
    value = 0
    if pos < n and text[pos] == "-":
        pos = pos + 1
        start = pos
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    if text[pos] == "0":
        pos = pos + 1
        if pos < n and text[pos] in _DIGIT_CHARS:
            raise JSONParseError(f"bad number at {pos}")
    elif text[pos] in _DIGIT_CHARS:
        while pos < n and text[pos] in _DIGIT_CHARS:
            value = value * 10 + (ord(text[pos]) - 48)
            pos = pos + 1
    else:
        raise JSONParseError(f"bad number at {pos}")
    if pos < n and text[pos] == ".":
        is_float = True
        pos = pos + 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number fractional at {pos}")
        decimal = 0.1
        while pos < n and text[pos] in _DIGIT_CHARS:
            value += (ord(text[pos]) - 48) * decimal
            decimal *= 0.1
            pos = pos + 1
    if pos < n and text[pos] in ("e", "E"):
        is_float = True
        pos = pos + 1
        if pos < n and text[pos] in ("+", "-"):
            pos = pos + 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number exponent at {pos}")
        exp = 0
        while pos < n and text[pos] in _DIGIT_CHARS:
            exp = exp * 10 + (ord(text[pos]) - 48)
            pos = pos + 1

            value *= 10 ** exp
        else:
            value *= 10 ** exp

    return int(value), pos
