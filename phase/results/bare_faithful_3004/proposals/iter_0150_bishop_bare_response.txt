def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    start = pos
    # Handle optional sign
    if text[pos] == "-":
        pos += 1
    if pos >= len(text):
        raise JSONParseError(f"bad number at {start}")
    # Parse integer part
    if text[pos] == "0":
        pos += 1
    else:
        while pos < len(text) and text[pos] in _DIGIT_CHARS:
            pos += 1
    # Parse fractional part
    if pos < len(text) and text[pos] == ".":
        pos += 1
        if pos >= len(text) or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number at {start}")
        while pos < len(text) and text[pos] in _DIGIT_CHARS:
            pos += 1
    # Parse exponent
    if pos < len(text) and text[pos].lower() == "e":
        pos += 1
        if pos < len(text) and text[pos] in "+-":
            pos += 1
        if pos >= len(text) or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number at {start}")
        while pos < len(text) and text[pos] in _DIGIT_CHARS:
            pos += 1
    # Convert to appropriate type
    s = text[start:pos]
    if "." in s or "e" in s.lower():
        return float(s), pos
    return int(s), pos
