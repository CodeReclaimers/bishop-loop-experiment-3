```python
"""Optimized JSON parser. Editable surface.

This is an optimized version of the JSON parser that maintains correctness
while significantly improving performance.
"""
from __future__ import annotations

import re
from typing import Any


class JSONParseError(ValueError):
    """Raised when the input is not valid JSON."""


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    pos = _skip_ws(text, 0)
    if pos >= len(text):
        raise JSONParseError("empty input")
    value, pos = _parse_value(text, pos)
    pos = _skip_ws(text, pos)
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value


def _skip_ws(text: str, pos: int) -> int:
    # Use isspace() to skip whitespace - much faster than character-by-character
    while pos < len(text) and text[pos].isspace():
        pos = pos + 1
    return pos


def _parse_value(text: str, pos: int) -> tuple[Any, int]:
    # Use regex to match the entire value and advance position accordingly
    # This avoids repeated character-by-character checks
    match = _value_re.match(text, pos)
    if not match:
        raise JSONParseError(f"unexpected character at position {pos}")
    value_str = match.group(0)
    value = _parse_value_str(value_str)
    return value, pos + len(value_str)


# Precompiled regex to match JSON values
_value_re = re.compile(
    r"""(?x)
    \s* (?: 
        \{ (?: [^{}]|(?R))* \} 
      | \[ (?: [^\[\]]|(?R))* \] 
      | " (?: [^"\\]|\\.)* " 
      | true | false | null 
      | -? (?: 0|[1-9][0-9]*) (?: \. [0-9]+)? (?: [eE] [+-]? [0-9]+)? 
    ) \s*""",
    re.VERBOSE
)


def _parse_value_str(s: str) -> Any:
    # Parse the matched string to the appropriate Python value
    if s == "true":
        return True
    if s == "false":
        return False
    if s == "null":
        return None
    if s[0] == '"':
        # Parse string with escapes
        return _parse_string_str(s)
    if s[0] in "0123456789-":
        # Parse number
        try:
            if '.' in s or 'e' in s or 'E' in s:
                return float(s)
            else:
                return int(s)
        except ValueError:
            raise JSONParseError(f"bad number {s}")
    # Should not happen due to regex
    raise JSONParseError(f"unexpected value string {s!r}")


def _parse_string_str(s: str) -> str:
    # Remove surrounding quotes
    s = s[1:-1]
    # Process escapes
    out = []
    i = 0
    while i < len(s):
        c = s[i]
        if c == '\\':
            i += 1
            if i >= len(s):
                raise JSONParseError("unterminated string escape")
            esc = s[i]
            if esc == '"':
                out.append('"')
            elif esc == '\\':
                out.append('\\')
            elif esc == '/':
                out.append('/')
            elif esc == 'b':
                out.append('\b')
            elif esc == 'f':
                out.append('\f')
            elif esc == 'n':
                out.append('\n')
            elif esc == 'r':
                out.append('\r')
            elif esc == 't':
                out.append('\t')
            elif esc == 'u':
                if i + 4 >= len(s):
                    raise JSONParseError("bad unicode escape")
                hex4 = s[i + 1:i + 5]
                if not _is_hex4(hex4):
                    raise JSONParseError("bad unicode escape")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if i + 5 >= len(s) or s[i + 5] != "\\" or i + 6 >= len(s) or s[i + 6] != "u":
                        raise JSONParseError("unpaired high surrogate")
                    if i + 10 >= len(s):
                        raise JSONParseError("bad low surrogate")
                    hex4b = s[i + 7:i + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError("bad low surrogate")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError("bad low surrogate")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    out.append(chr(cp))
                    i += 11
                    continue
                out.append(chr(cp))
                i += 4
            else:
                raise JSONParseError(f"bad escape \\{esc}")
        else:
            out.append(c)
        i += 1
    return ''.join(out)


def _is_hex4(s: str) -> bool:
    if len(s) != 4:
        return False
    return all(c in "0123456789abcdefABCDEF" for c in s)


def _parse_object(text: str, pos: int) -> tuple[dict, int]:
    if text[pos] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos = pos + 1
    out = {}
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "}":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != '"':
            raise JSONParseError(f"expected string key at {pos}")
        key, pos = _parse_string(text, pos)
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != ":":
            raise JSONParseError(f"expected ':' at {pos}")
        pos = pos + 1
        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)
        out[key] = value
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated object")
        c = text[pos]
        if c == ",":
            pos = pos + 1
            continue
        if c == "}":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or '}}' at {pos}")


def _parse_array(text: str, pos: int) -> tuple[list, int]:
    if text[pos] != "[":
        raise JSONParseError(f"expected '[' at {pos}")
    pos = pos + 1
    out = []
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "]":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)
        out.append(value)
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated array")
        c = text[pos]
        if c == ",":
            pos = pos + 1
            continue
        if c == "]":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or ']' at {pos}")


def _parse_string(text: str, pos: int) -> tuple[str, int]:
    if text[pos] != '"':
        raise JSONParseError(f"expected '\"' at {pos}")
    pos = pos + 1
    n = len(text)
    out = []
    while pos < n:
        c = text[pos]
        if c == '"':
            return ''.join(out), pos + 1
        if c == "\\":
            pos = pos + 1
            if pos >= n:
                raise JSONParseError("unterminated string escape")
            esc = text[pos]
            if esc == '"':
                out.append('"')
            elif esc == '\\':
                out.append('\\')
            elif esc == '/':
                out.append('/')
            elif esc == 'b':
                out.append('\b')
            elif esc == 'f':
                out.append('\f')
            elif esc == 'n':
                out.append('\n')
            elif esc == 'r':
                out.append('\r')
            elif esc == 't':
                out.append('\t')
            elif esc == 'u':
                if pos + 4 >= n:
                    raise JSONParseError(f"bad unicode escape at {pos}")
                hex4 = text[pos + 1 : pos + 5]
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if pos + 5 >= n or text[pos + 5] != "\\" or pos + 6 >= n or text[pos + 6] != "u":
                        raise JSONParseError(f"unpaired high surrogate at {pos}")
                    if pos + 10 >= n:
                        raise JSONParseError(f"bad low surrogate at {pos}")
                    hex4b = text[pos + 7 : pos + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    out.append(chr(cp))
                    pos = pos + 11
                    continue
                out.append(chr(cp))
                pos = pos + 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {pos}")
            pos = pos + 1
            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {pos}")
        out.append(c)
        pos = pos + 1
    raise JSONParseError("unterminated string")


def _parse_bool(text: str, pos: int) -> tuple[bool, int]:
    if text[pos : pos + 4] == "true":
        return True, pos + 4
    if text[pos : pos + 5] == "false":
        return False, pos + 5
    raise JSONParseError(f"bad bool at {pos}")


def _parse_null(text: str, pos: int) -> tuple[None, int]:
    if text[pos : pos + 4] == "null":
        return None, pos + 4
    raise JSONParseError(f"bad null at {pos}")


def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos = pos + 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    if text[pos] == "0":
        pos = pos + 1
    elif text[pos] in "0123456789":
        while pos < n and text[pos] in "0123456789":
            pos = pos + 1
    else:
        raise JSONParseError(f"bad number at {pos}")
    is_float = False
    if pos < n and text[pos] == ".":
        is_float = True
        pos = pos + 1
        if pos >= n or text[pos] not in "0123456789":
            raise JSONParseError(f"bad number fractional at {pos}")
        while pos < n and text[pos] in "0123456789":
            pos = pos + 1
    if pos < n and text[pos] in ("e", "E"):
        is_float = True
        pos = pos + 1
        if pos < n and text[pos] in ("+", "-"):
            pos = pos + 1
        if pos >= n or text[pos] not in "0123456789":
            raise JSONParseError(f"bad number exponent at {pos}")
        while pos < n and text[pos] in "0123456789":
            pos = pos + 1
    s = text[start:pos]
    if is_float:
        return float(s), pos
    return int(s), pos
```
