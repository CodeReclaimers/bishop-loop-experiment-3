```python
"""Optimized JSON parser. Editable surface.

This is an optimized version of the JSON parser that maintains correctness
while significantly improving performance.
"""
from __future__ import annotations

from typing import Any


class JSONParseError(ValueError):
    """Raised when the input is not valid JSON."""


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    iterator = _StringIterator(text)
    pos = _skip_ws(iterator)
    if pos >= len(text):
        raise JSONParseError("empty input")
    value, pos = _parse_value(iterator, pos)
    pos = _skip_ws(iterator, pos)
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value


class _StringIterator:
    def __init__(self, text: str):
        self.text = text
        self.pos = 0

    def __getitem__(self, index: int) -> str:
        return self.text[index]

    def __len__(self) -> int:
        return len(self.text)

    def peek(self, offset: int = 0) -> str:
        pos = self.pos + offset
        if pos >= len(self.text):
            return ""
        return self.text[pos]

    def advance(self, count: int = 1) -> None:
        self.pos += count


def _skip_ws(iterator: _StringIterator, pos: int = None) -> int:
    if pos is None:
        pos = iterator.pos
    while pos < len(iterator) and iterator[pos].isspace():
        pos = pos + 1
    return pos


def _parse_value(iterator: _StringIterator, pos: int) -> tuple[Any, int]:
    pos = _skip_ws(iterator, pos)
    if pos >= len(iterator):
        raise JSONParseError("unexpected end of input")
    c = iterator[pos]
    if c == "{":
        return _parse_object(iterator, pos)
    if c == "[":
        return _parse_array(iterator, pos)
    if c == '"':
        return _parse_string(iterator, pos)
    if c == "t" or c == "f":
        return _parse_bool(iterator, pos)
    if c == "n":
        return _parse_null(iterator, pos)
    if c == "-" or c in "0123456789":
        return _parse_number(iterator, pos)
    raise JSONParseError(f"unexpected character {c!r} at position {pos}")


def _parse_object(iterator: _StringIterator, pos: int) -> tuple[dict, int]:
    if iterator[pos] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos = pos + 1
    out = {}
    pos = _skip_ws(iterator, pos)
    if pos < len(iterator) and iterator[pos] == "}":
        return out, pos + 1
    while True:
        pos = _skip_ws(iterator, pos)
        if pos >= len(iterator) or iterator[pos] != '"':
            raise JSONParseError(f"expected string key at {pos}")
        key, pos = _parse_string(iterator, pos)
        pos = _skip_ws(iterator, pos)
        if pos >= len(iterator) or iterator[pos] != ":":
            raise JSONParseError(f"expected ':' at {pos}")
        pos = pos + 1
        pos = _skip_ws(iterator, pos)
        value, pos = _parse_value(iterator, pos)
        out[key] = value
        pos = _skip_ws(iterator, pos)
        if pos >= len(iterator):
            raise JSONParseError("unterminated object")
        c = iterator[pos]
        if c == ",":
            pos = pos + 1
            continue
        if c == "}":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or '}}' at {pos}")


def _parse_array(iterator: _StringIterator, pos: int) -> tuple[list, int]:
    if iterator[pos] != "[":
        raise JSONParseError(f"expected '[' at {pos}")
    pos = pos + 1
    out = []
    pos = _skip_ws(iterator, pos)
    if pos < len(iterator) and iterator[pos] == "]":
        return out, pos + 1
    while True:
        pos = _skip_ws(iterator, pos)
        value, pos = _parse_value(iterator, pos)
        out.append(value)
        pos = _skip_ws(iterator, pos)
        if pos >= len(iterator):
            raise JSONParseError("unterminated array")
        c = iterator[pos]
        if c == ",":
            pos = pos + 1
            continue
        if c == "]":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or ']' at {pos}")


def _parse_string(iterator: _StringIterator, pos: int) -> tuple[str, int]:
    if iterator[pos] != '"':
        raise JSONParseError(f"expected '\"' at {pos}")
    pos = pos + 1
    n = len(iterator)
    out = []
    while pos < n:
        c = iterator[pos]
        if c == '"':
            return ''.join(out), pos + 1
        if c == "\\":
            pos = pos + 1
            if pos >= n:
                raise JSONParseError("unterminated string escape")
            esc = iterator[pos]
            if esc == '"':
                out.append('"')
            elif esc == '\\':
                out.append('\\')
            elif esc == '/':
                out.append('/')
            elif esc == 'b':
                out.append('\b')
            elif esc == 'f':
                out.append('\f')
            elif esc == 'n':
                out.append('\n')
            elif esc == 'r':
                out.append('\r')
            elif esc == 't':
                out.append('\t')
            elif esc == 'u':
                if pos + 4 >= n:
                    raise JSONParseError(f"bad unicode escape at {pos}")
                hex4 = iterator[pos + 1 : pos + 5]
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if pos + 5 >= n or iterator[pos + 5] != "\\" or pos + 6 >= n or iterator[pos + 6] != "u":
                        raise JSONParseError(f"unpaired high surrogate at {pos}")
                    if pos + 10 >= n:
                        raise JSONParseError(f"bad low surrogate at {pos}")
                    hex4b = iterator[pos + 7 : pos + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    out.append(chr(cp))
                    pos = pos + 11
                    continue
                out.append(chr(cp))
                pos = pos + 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {pos}")
            pos = pos + 1
            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {pos}")
        out.append(c)
        pos = pos + 1
    raise JSONParseError("unterminated string")


def _is_hex4(s: str) -> bool:
    if len(s) != 4:
        return False
    return all(c in "0123456789abcdefABCDEF" for c in s)


def _parse_bool(iterator: _StringIterator, pos: int) -> tuple[bool, int]:
    if iterator[pos : pos + 4] == "true":
        return True, pos + 4
    if iterator[pos : pos + 5] == "false":
        return False, pos + 5
    raise JSONParseError(f"bad bool at {pos}")


def _parse_null(iterator: _StringIterator, pos: int) -> tuple[None, int]:
    if iterator[pos : pos + 4] == "null":
        return None, pos + 4
    raise JSONParseError(f"bad null at {pos}")


def _parse_number(iterator: _StringIterator, pos: int) -> tuple[float | int, int]:
    n = len(iterator)
    start = pos
    if pos < n and iterator[pos] == "-":
        pos = pos + 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    if iterator[pos] == "0":
        pos = pos + 1
    elif iterator[pos] in "0123456789":
        while pos < n and iterator[pos] in "0123456789":
            pos = pos + 1
    else:
        raise JSONParseError(f"bad number at {pos}")
    is_float = False
    if pos < n and iterator[pos] == ".":
        is_float = True
        pos = pos + 1
        if pos >= n or iterator[pos] not in "0123456789":
            raise JSONParseError(f"bad number fractional at {pos}")
        while pos < n and iterator[pos] in "0123456789":
            pos = pos + 1
    if pos < n and iterator[pos] in ("e", "E"):
        is_float = True
        pos = pos + 1
        if pos < n and iterator[pos] in ("+", "-"):
            pos = pos + 1
        if pos >= n or iterator[pos] not in "0123456789":
            raise JSONParseError(f"bad number exponent at {pos}")
        while pos < n and iterator[pos] in "0123456789":
            pos = pos + 1
    s = iterator.text[start:pos]
    if is_float:
        return float(s), pos
    return int(s), pos
```
