```python
"""Optimized JSON parser using regular expressions. Editable surface.

This is an optimized version of the JSON parser that maintains correctness
while significantly improving performance using regex patterns.
"""
from __future__ import annotations

import re
from typing import Any


class JSONParseError(ValueError):
    """Raised when the input is not valid JSON."""


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    
    # Precompile regex patterns
    whitespace = re.compile(r'\s*')
    string_pattern = re.compile(r'"((?:[^"\\]|\\.)*?)"')
    number_pattern = re.compile(r'(-?(?:0|[1-9]\d*)(?:\.\d+)?(?:[eE][+-]?\d+)?)')
    true_pattern = re.compile(r'true')
    false_pattern = re.compile(r'false')
    null_pattern = re.compile(r'null')
    
    def parse_value(text: str, pos: int) -> tuple[Any, int]:
        # Skip whitespace
        match = whitespace.match(text, pos)
        if not match:
            raise JSONParseError(f"unexpected end of input at position {pos}")
        pos = match.end()
        
        if pos >= len(text):
            raise JSONParseError("unexpected end of input")
        
        c = text[pos]
        if c == '{':
            return parse_object(text, pos)
        elif c == '[':
            return parse_array(text, pos)
        elif c == '"':
            match = string_pattern.match(text, pos)
            if not match:
                raise JSONParseError(f"invalid string at position {pos}")
            return match.group(1), match.end()
        elif c == 't':
            match = true_pattern.match(text, pos)
            if not match:
                raise JSONParseError(f"invalid true at position {pos}")
            return True, match.end()
        elif c == 'f':
            match = false_pattern.match(text, pos)
            if not match:
                raise JSONParseError(f"invalid false at position {pos}")
            return False, match.end()
        elif c == 'n':
            match = null_pattern.match(text, pos)
            if not match:
                raise JSONParseError(f"invalid null at position {pos}")
            return None, match.end()
        elif c == '-' or c in '0123456789':
            match = number_pattern.match(text, pos)
            if not match:
                raise JSONParseError(f"invalid number at position {pos}")
            num_str = match.group(1)
            if '.' in num_str or 'e' in num_str.lower():
                return float(num_str), match.end()
            else:
                return int(num_str), match.end()
        else:
            raise JSONParseError(f"unexpected character {c!r} at position {pos}")
    
    def parse_object(text: str, pos: int) -> tuple[dict, int]:
        if text[pos] != '{':
            raise JSONParseError(f"expected '{{' at {pos}")
        pos = pos + 1
        
        # Skip whitespace
        match = whitespace.match(text, pos)
        if not match:
            raise JSONParseError(f"unexpected end of input at position {pos}")
        pos = match.end()
        
        if pos < len(text) and text[pos] == '}':
            return {}, pos + 1
        
        out = {}
        while True:
            # Parse key
            match = string_pattern.match(text, pos)
            if not match:
                raise JSONParseError(f"expected string key at {pos}")
            key = match.group(1)
            pos = match.end()
            
            # Skip whitespace
            match = whitespace.match(text, pos)
            if not match:
                raise JSONParseError(f"unexpected end of input at position {pos}")
            pos = match.end()
            
            if pos >= len(text) or text[pos] != ':':
                raise JSONParseError(f"expected ':' at {pos}")
            pos = pos + 1
            
            # Skip whitespace
            match = whitespace.match(text, pos)
            if not match:
                raise JSONParseError(f"unexpected end of input at position {pos}")
            pos = match.end()
            
            # Parse value
            value, pos = parse_value(text, pos)
            out[key] = value
            
            # Skip whitespace
            match = whitespace.match(text, pos)
            if not match:
                raise JSONParseError(f"unexpected end of input at position {pos}")
            pos = match.end()
            
            if pos >= len(text):
                raise JSONParseError("unterminated object")
            
            c = text[pos]
            if c == ',':
                pos = pos + 1
                # Skip whitespace
                match = whitespace.match(text, pos)
                if not match:
                    raise JSONParseError(f"unexpected end of input at position {pos}")
                pos = match.end()
                continue
            elif c == '}':
                return out, pos + 1
            else:
                raise JSONParseError(f"expected ',' or '}}' at {pos}")
    
    def parse_array(text: str, pos: int) -> tuple[list, int]:
        if text[pos] != '[':
            raise JSONParseError(f"expected '[' at {pos}")
        pos = pos + 1
        
        # Skip whitespace
        match = whitespace.match(text, pos)
        if not match:
            raise JSONParseError(f"unexpected end of input at position {pos}")
        pos = match.end()
        
        if pos < len(text) and text[pos] == ']':
            return [], pos + 1
        
        out = []
        while True:
            # Parse value
            value, pos = parse_value(text, pos)
            out.append(value)
            
            # Skip whitespace
            match = whitespace.match(text, pos)
            if not match:
                raise JSONParseError(f"unexpected end of input at position {pos}")
            pos = match.end()
            
            if pos >= len(text):
                raise JSONParseError("unterminated array")
            
            c = text[pos]
            if c == ',':
                pos = pos + 1
                # Skip whitespace
                match = whitespace.match(text, pos)
                if not match:
                    raise JSONParseError(f"unexpected end of input at position {pos}")
                pos = match.end()
                continue
            elif c == ']':
                return out, pos + 1
            else:
                raise JSONParseError(f"expected ',' or ']' at {pos}")
    
    pos = 0
    value, pos = parse_value(text, pos)
    
    # Skip whitespace
    match = whitespace.match(text, pos)
    if not match:
        raise JSONParseError(f"unexpected end of input at position {pos}")
    pos = match.end()
    
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    
    return value
```
