This is the starting point for the optimization loop. It is correct against
`json.loads` on the full corpus but is intentionally written in a way that
leaves significant performance on the table:

- Whitespace is skipped via a per-call Python loop instead of a regex.
- Numbers are scanned character-by-character with explicit predicates.
- Object/array building uses string concatenation and rebuilt-on-each-step
  collection (`out = out + [v]`, `out = dict(out); out[k] = v`).
- Recursive descent with redundant whitespace handling at every level.

The optimization target is wall-clock time to parse the 200-case fixed

point and must raise `JSONParseError` for malformed inputs.

_WS_CHARS = frozenset(" \t\n\r")
_DIGIT_CHARS = frozenset("0123456789")


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    pos = _skip_ws(text, 0)
    if pos >= len(text):
        raise JSONParseError("empty input")
    value, pos = _parse_value(text, pos)
    pos = _skip_ws(text, pos)
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value


def _skip_ws(text: str, pos: int) -> int:
    n = len(text)
    while pos < n and text[pos] in _WS_CHARS:
        pos += 1
    return pos


def _parse_value(text: str, pos: int) -> tuple[Any, int]:

    if pos >= len(text):

    if c == "[":
        return _parse_array(text, pos)
    if c == '"':
        return _parse_string(text, pos)
    if c == "t" or c == "f":
        return _parse_bool(text, pos)
    if c == "n":
        return _parse_null(text, pos)
    if c == "-" or c in _DIGIT_CHARS:
        return _parse_number(text, pos)
    raise JSONParseError(f"unexpected character {c!r} at position {pos}")

`json.loads` on the full corpus but is intentionally written in a way that
leaves significant performance on the table:

- Numbers are scanned character-by-character with explicit predicates.
- Object/array building uses string concatenation and rebuilt-on-each-step
  collection (`out = out + [v]`, `out = dict(out); out[k] = v`).

point and must raise `JSONParseError` for malformed inputs.

from typing import Any

    """Raised when the input is not valid JSON."""

_DIGIT_CHARS = frozenset("0123456789")


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    
    # FSM-based parsing
    class State:
        START = 0
        IN_OBJECT = 1
        IN_ARRAY = 2
        IN_STRING = 3
        IN_NUMBER = 4
        IN_TRUE = 5
        IN_FALSE = 6
        IN_NULL = 7
        SKIP_WS = 8
        DONE = 9
    
    # Simple state machine implementation
    pos = 0
    state = State.START
    stack = []
    current_object = {}
    current_array = []
    current_key = ""
    current_string = ""
    current_number = ""
    expecting_key = False
    expecting_value = False
    
    def consume_char():
        nonlocal pos
        if pos >= len(text):
            raise JSONParseError("unexpected end of input")
        c = text[pos]
        pos += 1
        return c
    
    def peek_char():
        if pos >= len(text):
            return None
        return text[pos]
    
    def skip_whitespace():
        nonlocal pos
        while pos < len(text) and text[pos] in _WS_CHARS:
            pos += 1
    
    def parse_value():
        nonlocal pos, state, current_object, current_array, current_key, current_string, current_number, expecting_key, expecting_value
        
        skip_whitespace()
        if pos >= len(text):
            raise JSONParseError("unexpected end of input")
        
        c = text[pos]
        if c == '{':
            pos += 1
            stack.append(current_object)
            current_object = {}
            expecting_key = True
            return parse_value()
        elif c == '[':
            pos += 1
            stack.append(current_array)
            current_array = []
            expecting_value = True
            return parse_value()
        elif c == '"':
            pos += 1
            current_string = ""
            while pos < len(text):
                c = text[pos]
                pos += 1
                if c == '"':
                    break
                elif c == '\\':
                    if pos >= len(text):
                        raise JSONParseError("unterminated string escape")
                    esc = text[pos]
                    pos += 1
                    if esc == '"':
                        current_string += '"'
                    elif esc == '\\':
                        current_string += "\\"
                    elif esc == '/':
                        current_string += "/"
                    elif esc == 'b':
                        current_string += "\b"
                    elif esc == 'f':
                        current_string += "\f"
                    elif esc == 'n':
                        current_string += "\n"
                    elif esc == 'r':
                        current_string += "\r"
                    elif esc == 't':
                        current_string += "\t"
                    elif esc == 'u':
                        if pos + 4 >= len(text):
                            raise JSONParseError(f"bad unicode escape at {pos}")
                        hex4 = text[pos : pos + 4]
                        if not _is_hex4(hex4):
                            raise JSONParseError(f"bad unicode escape at {pos}")
                        cp = int(hex4, 16)
                        if 0xD800 <= cp <= 0xDBFF:
                            if pos + 5 >= len(text) or text[pos + 5] != '\\' or pos + 6 >= len(text) or text[pos + 6] != 'u':
                                raise JSONParseError(f"unpaired high surrogate at {pos}")
                            if pos + 10 >= len(text):
                                raise JSONParseError(f"bad low surrogate at {pos}")
                            hex4b = text[pos + 7 : pos + 11]
                            if not _is_hex4(hex4b):
                                raise JSONParseError(f"bad low surrogate at {pos + 7}")
                            cp2 = int(hex4b, 16)
                            if not (0xDC00 <= cp2 <= 0xDFFF):
                                raise JSONParseError(f"bad low surrogate at {pos + 7}")
                            cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                            current_string += chr(cp)
                            pos += 11
                            continue
                        current_string += chr(cp)
                        pos += 4
                elif ord(c) < 0x20:
                    raise JSONParseError(f"unescaped control character at {pos}")
                else:
                    current_string += c
            return current_string
        elif c == 't':
            if text[pos:pos+4] == "true":
                pos += 4
                return True
            else:
                raise JSONParseError(f"bad bool at {pos}")
        elif c == 'f':
            if text[pos:pos+5] == "false":
                pos += 5
                return False
            else:
                raise JSONParseError(f"bad bool at {pos}")
        elif c == 'n':
            if text[pos:pos+4] == "null":
                pos += 4
                return None
            else:
                raise JSONParseError(f"bad null at {pos}")
        elif c == '-' or c in _DIGIT_CHARS:
            current_number = ""
            while pos < len(text) and (text[pos] in _DIGIT_CHARS or text[pos] in ".eE+-"):
                current_number += text[pos]
                pos += 1
            if '.' in current_number or 'e' in current_number.lower():
                return float(current_number)
            return int(current_number)
        else:
            raise JSONParseError(f"unexpected character {c!r} at position {pos}")
    
    # Parse the top-level value
    result = parse_value()
    
    # Skip trailing whitespace
    skip_whitespace()
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    
    return result
