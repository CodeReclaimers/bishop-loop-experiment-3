def _is_hex4(s: str) -> bool:
    if len(s) != 4:
        return False
    # Fast bitmask check for hex characters
    mask = 0
    for ch in s:
        if ch in _HEX_CHARS:
            # Set bit for this hex character
            if '0' <= ch <= '9':
                mask |= 1 << (ord(ch) - ord('0'))
            elif 'A' <= ch <= 'F':
                mask |= 1 << (ord(ch) - ord('A') + 10)
            elif 'a' <= ch <= 'f':
                mask |= 1 << (ord(ch) - ord('a') + 10)
            else:
                return False
        else:
            return False
    # Check if all bits are valid hex bits (0-9, A-F, a-f)
    return mask == 0x3FF  # 1111111111 in binary (10 bits for 0-9, A-F, a-f)
