import re

# Precompile the number regex pattern for performance
_NUMBER_RE = re.compile(r"""
    -?          # optional minus sign
    (?:0|[1-9]\d*)  # integer part (0 or non-zero start)
    (?:\.\d+)?  # optional fractional part
    (?:[eE][+-]?\d+)?  # optional exponent part
""", re.VERBOSE)

def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    # Find the longest match starting at pos
    match = _NUMBER_RE.match(text, pos)
    if not match:
        raise JSONParseError(f"bad number at {pos}")
