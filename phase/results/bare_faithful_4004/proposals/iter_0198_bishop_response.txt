To optimize the `parse` function in `json_parser.py`, we can use a more efficient way to handle string escaping within JSON values. Specifically, we can use a loop instead of regex and a conditional check for each escape character. This approach reduces the overhead associated with regex matching and avoids unnecessary allocations, resulting in faster parsing times.

Here's how you can implement this change:

```python
def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    pos = _skip_ws(text, 0)
    if pos >= len(text):
        raise JSONParseError("empty input")
    value, pos = _parse_value(text, pos)
    pos = _skip_ws(text, pos)
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value

def _skip_ws(text: str, pos: int) -> int:
    while pos < len(text) and text[pos] in " \t\n":
        pos += 1
    return pos


def _parse_value(text: str, pos: int) -> tuple[Any, int]:
    pos = _skip_ws(text, pos)
    if pos >= len(text):
        raise JSONParseError("unexpected end of input")
    c = text[pos]
    if c == "{":
        return _parse_object(text, pos)
    if c == "[":
        return _parse_array(text, pos)
    if c == '"':
        return _parse_string(text, pos)
    if c == "t" or c == "f":
        return _parse_bool(text, pos)

def _parse_string(text: str, pos: int) -> tuple[str, int]:
    out = ""
    while True:
        if pos >= len(text):
            raise JSONParseError("unterminated string")
        c = text[pos]
        if c == '"':
            break
        if c == "\\":
            pos += 1
            if pos >= len(text):
                raise JSONParseError("unterminated string escape")
            esc = text[pos]
            if esc in _ESCAPE_MAP:
                out += _ESCAPE_MAP[esc]
            elif esc == "u":
                if pos + 4 >= len(text):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                hex4 = text[pos + 1 : pos + 5]
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if pos + 5 >= len(text) or text[pos + 5] != "\\" or pos + 6 >= len(text) or text[pos + 6] != "u":
                        raise JSONParseError(f"unpaired high surrogate at {pos}")
                    if pos + 10 >= len(text):
                        raise JSONParseError(f"bad low surrogate at {pos}")
                    hex4b = text[pos + 7 : pos + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    out += chr(cp)
                    pos = pos + 11
                    continue
                out += chr(cp)
                pos = pos + 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {pos}")
            pos = pos + 1
            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {pos}")
        out += c
        pos = pos + 1
    return out, pos
```

In this change, we introduce a loop to parse the string characters. When we encounter an escape sequence (`\\`), we read the next four characters to form the Unicode code point and decode it using `chr()`. This approach avoids the overhead of regex matching and reduces the number of conditional checks required during parsing.
