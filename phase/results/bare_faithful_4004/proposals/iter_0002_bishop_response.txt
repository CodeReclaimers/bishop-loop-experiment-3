To optimize the JSON parser, consider using an iterative approach with index-based iteration rather than regex for whitespace skipping. This can reduce overhead from regex operations. Additionally, avoid redundant string concatenation by directly appending to a list in `parse_string`. The optimized version would look something like this:

```python
def _skip_ws(text: str, pos: int) -> int:
    while pos < len(text) and text[pos] in " \t\n":
        pos += 1
    return pos

def _parse_string(text: str, pos: int) -> tuple[str, int]:
    if text[pos] != '"':
        raise JSONParseError(f"expected string at {pos}")
    out = []
    while True:
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated string")
        c = text[pos]
        if c == '"':
            return ''.join(out), pos + 1
        if c == "\\":
            pos = pos + 1
            if pos >= len(text):
                raise JSONParseError("unterminated string escape")
            esc = text[pos]
            if esc in _ESCAPE_MAP:
                out.append(_ESCAPE_MAP[esc])
            elif esc == "u":
                if pos + 4 >= len(text):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                hex4 = text[pos + 1 : pos + 5]
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if pos + 5 >= len(text) or text[pos + 5] != "\\" or pos + 6 >= len(text) or text[pos + 6] != "u":
                        raise JSONParseError(f"unpaired high surrogate at {pos}")
                    if pos + 10 >= len(text):
                        raise JSONParseError(f"bad low surrogate at {pos}")
                    hex4b = text[pos + 7 : pos + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    out.append(chr(cp))
                    pos = pos + 11
                    continue
                out.append(chr(cp))
                pos = pos + 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {pos}")
            pos = pos + 1
            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {pos}")
        out.append(c)
        pos = pos + 1
    raise JSONParseError("unterminated string")
```

This approach reduces regex usage and improves performance by directly appending characters to the list as they are encountered, avoiding unnecessary string concatenations.
