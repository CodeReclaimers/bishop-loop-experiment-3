def _parse_object(text: str, pos: int) -> tuple[dict, int]:
    if text[pos] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos = pos + 1
    out = {}
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "}":
        return out, pos + 1
    
    # Stack-based approach for parsing object
    stack = [out]
    key = None
    expecting_key = True
    
    while stack:
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated object")
        
        c = text[pos]
        if expecting_key:
            if c == '"':
                key, pos = _parse_string(text, pos)
                expecting_key = False
            elif c == '}':
                stack.pop()
                if not stack:
                    return out, pos + 1
                expecting_key = False
                pos = pos + 1
                continue
            else:
                raise JSONParseError(f"expected string key at {pos}")
        else:
            if c == ':':
                pos = pos + 1
                pos = _skip_ws(text, pos)
                value, pos = _parse_value(text, pos)
                stack[-1][key] = value
                expecting_key = True
            elif c == ',':
                pos = pos + 1
                expecting_key = True
            elif c == '}':
                stack.pop()
                if not stack:
                    return out, pos + 1
                expecting_key = False
                pos = pos + 1
            else:
                raise JSONParseError(f"expected ':' or '}}' at {pos}")
    
    return out, pos
