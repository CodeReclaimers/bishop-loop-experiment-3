def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    # State machine for number parsing
    # 0: initial state, expecting sign or digit
    # 1: after sign, expecting digit or decimal point
    # 2: parsing integer part
    # 3: parsing fractional part
    # 4: parsing exponent part
    # 5: parsing exponent sign
    # 6: parsing exponent digits
    state = 0
    is_float = False
    while pos < n:
        c = text[pos]
        if state == 0:
            if c == "-":
                state = 1
            elif c == "0":
                state = 2
            elif c in _DIGIT_CHARS:
                state = 2
            else:
                raise JSONParseError(f"bad number at {pos}")
        elif state == 1:
            if c == "0":
                state = 2
            elif c in _DIGIT_CHARS:
                state = 2
            else:
                raise JSONParseError(f"bad number at {pos}")
        elif state == 2:
            if c in _DIGIT_CHARS:
                pos += 1
                continue
            elif c == ".":
                is_float = True
                state = 3
            elif c in ("e", "E"):
                is_float = True
                state = 4
            else:
                break
        elif state == 3:
            if c in _DIGIT_CHARS:
                state = 3
            else:
                break
        elif state == 4:
            if c in ("+", "-"):
                state = 5
            elif c in _DIGIT_CHARS:
                state = 6
            else:
                raise JSONParseError(f"bad number exponent at {pos}")
        elif state == 5:
            if c in _DIGIT_CHARS:
                state = 6
            else:
                raise JSONParseError(f"bad number exponent at {pos}")
        elif state == 6:
            if c in _DIGIT_CHARS:
                state = 6
            else:
                break
        pos += 1
    s = text[start:pos]
    if is_float:

    return int(s), pos
