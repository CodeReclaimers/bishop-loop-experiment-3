def _skip_ws(text: str, pos: int) -> int:
    n = len(text)
    # Binary search for the end of whitespace
    start = pos
    while pos < n and text[pos] in _WS_CHARS:
        pos += 1
    # Perform binary search to skip consecutive whitespace
    if pos > start:
        left = start
        right = pos - 1
        while left <= right:
            mid = (left + right) // 2
            if text[mid] in _WS_CHARS:
                left = mid + 1
            else:
                right = mid - 1
        return left
    return pos
