def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    # State machine for number parsing
    state = 0  # 0: integer part, 1: fraction part, 2: exponent part
    while pos < n:
        c = text[pos]
        if state == 0:
            if c == "0":
                pos += 1
            elif c in _DIGIT_CHARS:
                pos += 1
                while pos < n and text[pos] in _DIGIT_CHARS:
                    pos += 1
                if pos >= n:
                    break
                c = text[pos]
            else:
                raise JSONParseError(f"bad number at {pos}")
            if c == ".":
                state = 1
                pos += 1
            elif c in ("e", "E"):
                state = 2
                pos += 1
                if pos < n and text[pos] in ("+", "-"):
                    pos += 1
            else:
                break
        elif state == 1:
            if c in _DIGIT_CHARS:
                pos += 1
                while pos < n and text[pos] in _DIGIT_CHARS:
                    pos += 1
                if pos >= n:
                    break
                c = text[pos]
            else:

            if c in ("e", "E"):
                state = 2
                pos += 1
                if pos < n and text[pos] in ("+", "-"):
                    pos += 1
            else:
                break

            if c in _DIGIT_CHARS:
                pos += 1

                    pos += 1
                break
            else:

    s = text[start:pos]
    # Determine if it's a float or int
    is_float = (state == 1) or (state == 2)
