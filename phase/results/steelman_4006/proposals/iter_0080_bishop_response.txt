```python
# (in json_parser.py)
def _skip_ws(text: str, pos: int) -> int:
    # Fast whitespace skipping using direct character checks with bytes
    while pos < len(text):
        c = text[pos]
        if isinstance(c, bytes) and ord(c) in [ord(' '), ord('\t'), ord('\n'), ord('\r'), ord('\f')]:
            pos += 1
        else:
            break
    return pos
```
