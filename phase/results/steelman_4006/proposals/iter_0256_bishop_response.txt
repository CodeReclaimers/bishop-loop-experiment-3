def _skip_ws(text: str, pos: int) -> int:
    # Fast whitespace skipping using ord() for direct character checks
    while pos < len(text):
        c = text[pos]
        if ord(c) in (ord(' '), ord('\t'), ord('\n'), ord('\r'), ord('\f')):

        else:
            break
    return pos
