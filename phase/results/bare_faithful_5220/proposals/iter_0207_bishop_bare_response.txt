"""Deliberately slow but correct JSON parser. Editable surface.

This is the starting point for the optimization loop. It is correct against
`json.loads` on the full corpus but is intentionally written in a way that
leaves significant performance on the table:

- Whitespace is skipped via a per-call Python loop instead of a regex.
- Numbers are scanned character-by-character with explicit predicates.
- Object/array building uses string concatenation and rebuilt-on-each-step
  collection (`out = out + [v]`, `out = dict(out); out[k] = v`).
- Recursive descent with redundant whitespace handling at every level.

The optimization target is wall-clock time to parse the 200-case fixed
corpus. The candidate must keep `parse(text) -> Any` as the public entry
point and must raise `JSONParseError` for malformed inputs.
"""
from __future__ import annotations

from typing import Any


class JSONParseError(ValueError):
    """Raised when the input is not valid JSON."""


# Deliberately use lists not sets — every membership check is O(k) instead of O(1).
_WS_CHARS = frozenset(" \t\n\r")
_DIGIT_CHARS = frozenset("0123456789")


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    pos = _skip_ws(text, 0)
    if pos >= len(text):
        raise JSONParseError("empty input")
    value, pos = _parse_value(text, pos)
    pos = _skip_ws(text, pos)
    if pos != len(text):
        raise JSONParseError(f"trailing content at position {pos}")
    return value


def _skip_ws(text: str, pos: int) -> int:
    n = len(text)
    while pos < n and text[pos] in _WS_CHARS:
        pos += 1
    return pos


def _parse_value(text: str, pos: int) -> tuple[Any, int]:
    pos = _skip_ws(text, pos)
    if pos >= len(text):
        raise JSONParseError("unexpected end of input")
    c = text[pos]
    if c == "{":
        return _parse_object(text, pos)
    if c == "[":
        return _parse_array(text, pos)
    if c == '"':
        return _parse_string(text, pos)
    if c == "t" or c == "f":
        return _parse_bool(text, pos)
    if c == "n":
        return _parse_null(text, pos)
    if c == "-" or c in _DIGIT_CHARS:
        return _parse_number(text, pos)
    raise JSONParseError(f"unexpected character {c!r} at position {pos}")


def _parse_object(text: str, pos: int) -> tuple[dict, int]:
    if text[pos] != "{":
        raise JSONParseError(f"expected '{{' at {pos}")
    pos += 1
    out: dict = {}
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "}":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != '"':
            raise JSONParseError(f"expected string key at {pos}")
        key, pos = _parse_string(text, pos)
        pos = _skip_ws(text, pos)
        if pos >= len(text) or text[pos] != ":":
            raise JSONParseError(f"expected ':' at {pos}")
        pos += 1
        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)
        # Use direct dict update to avoid rebuilding
        out[key] = value
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated object")
        c = text[pos]
        if c == ",":
            pos += 1
            continue
        if c == "}":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or '}}' at {pos}")


def _parse_array(text: str, pos: int) -> tuple[list, int]:
    if text[pos] != "[":
        raise JSONParseError(f"expected '[' at {pos}")
    pos += 1
    out: list = []
    pos = _skip_ws(text, pos)
    if pos < len(text) and text[pos] == "]":
        return out, pos + 1
    while True:
        pos = _skip_ws(text, pos)
        value, pos = _parse_value(text, pos)
        # Use direct list append to avoid rebuilding
        out.append(value)
        pos = _skip_ws(text, pos)
        if pos >= len(text):
            raise JSONParseError("unterminated array")
        c = text[pos]
        if c == ",":
            pos += 1
            continue
        if c == "]":
            return out, pos + 1
        raise JSONParseError(f"expected ',' or ']' at {pos}")


def _parse_string(text: str, pos: int) -> tuple[str, int]:
    if text[pos] != '"':
        raise JSONParseError(f"expected '\"' at {pos}")
    pos = pos + 1
    n = len(text)
    out = ""
    while pos < n:
        c = text[pos]
        if c == '"':
            return out, pos + 1
        if c == "\\":
            pos = pos + 1
            if pos >= n:
                raise JSONParseError("unterminated string escape")
            esc = text[pos]
            if esc == '"':
                out = out + '"'
            elif esc == "\\":
                out = out + "\\"
            elif esc == "/":
                out = out + "/"
            elif esc == "b":
                out = out + "\b"
            elif esc == "f":
                out = out + "\f"
            elif esc == "n":
                out = out + "\n"
            elif esc == "r":
                out = out + "\r"
            elif esc == "t":
                out = out + "\t"
            elif esc == "u":
                if pos + 4 >= n:
                    raise JSONParseError(f"bad unicode escape at {pos}")
                hex4 = text[pos + 1 : pos + 5]
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if pos + 5 >= n or text[pos + 5] != "\\" or pos + 6 >= n or text[pos + 6] != "u":
                        raise JSONParseError(f"unpaired high surrogate at {pos}")
                    if pos + 10 >= n:
                        raise JSONParseError(f"bad low surrogate at {pos}")
                    hex4b = text[pos + 7 : pos + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    out = out + chr(cp)
                    pos = pos + 11
                    continue
                out = out + chr(cp)
                pos = pos + 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {pos}")
            pos = pos + 1
            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {pos}")
        out = out + c
        pos = pos + 1
    raise JSONParseError("unterminated string")


def _is_hex4(s: str) -> bool:
    if len(s) != 4:
        return False
    for ch in s:
        if not (("0" <= ch <= "9") or ("a" <= ch <= "f") or ("A" <= ch <= "F")):
            return False
    return True


def _parse_bool(text: str, pos: int) -> tuple[bool, int]:
    if text[pos : pos + 4] == "true":
        return True, pos + 4
    if text[pos : pos + 5] == "false":
        return False, pos + 5
    raise JSONParseError(f"bad bool at {pos}")


def _parse_null(text: str, pos: int) -> tuple[None, int]:
    if text[pos : pos + 4] == "null":
        return None, pos + 4
    raise JSONParseError(f"bad null at {pos}")


def _parse_number(text: str, pos: int) -> tuple[float | int, int]:
    n = len(text)
    start = pos
    if pos < n and text[pos] == "-":
        pos += 1
    if pos >= n:
        raise JSONParseError(f"bad number at {pos}")
    if text[pos] == "0":
        pos += 1
    elif text[pos] in _DIGIT_CHARS:
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    else:
        raise JSONParseError(f"bad number at {pos}")
    is_float = False
    if pos < n and text[pos] == ".":
        is_float = True
        pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number fractional at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    if pos < n and text[pos] in ("e", "E"):
        is_float = True
        pos += 1
        if pos < n and text[pos] in ("+", "-"):
            pos += 1
        if pos >= n or text[pos] not in _DIGIT_CHARS:
            raise JSONParseError(f"bad number exponent at {pos}")
        while pos < n and text[pos] in _DIGIT_CHARS:
            pos += 1
    s = text[start:pos]
    if is_float:
        return float(s), pos
    return int(s), pos

This is the starting point for the optimization loop. It is correct against
`json.loads` on the full corpus but is intentionally written in a way that
leaves significant performance on the table:

- Whitespace is skipped via a per-call Python loop instead of a regex.
- Numbers are scanned character-by-character with explicit predicates.
- Object/array building uses string concatenation and rebuilt-on-each-step
  collection (`out = out + [v]`, `out = dict(out); out[k] = v`).
- Recursive descent with redundant whitespace handling at every level.

The optimization target is wall-clock time to parse the 200-case fixed
corpus. The candidate must keep `parse(text) -> Any` as the public entry
point and must raise `JSONParseError` for malformed inputs.
"""
from __future__ import annotations

from typing import Any


class JSONParseError(ValueError):
    """Raised when the input is not valid JSON."""


# Deliberately use lists not sets — every membership check is O(k) instead of O(1).
_WS_CHARS = frozenset(" \t\n\r")
_DIGIT_CHARS = frozenset("0123456789")


def parse(text: str) -> Any:
    if not isinstance(text, str):
        raise JSONParseError("input must be a string")
    it = _CharIterator(text)
    value, _ = _parse_value(it)
    if not it.eof():
        raise JSONParseError(f"trailing content at position {it.pos}")
    return value


class _CharIterator:
    def __init__(self, text: str):
        self.text = text
        self.pos = 0

    def peek(self) -> str:
        if self.pos >= len(self.text):
            return ""
        return self.text[self.pos]

    def advance(self) -> str:
        if self.pos >= len(self.text):
            return ""
        c = self.text[self.pos]
        self.pos += 1
        return c

    def eof(self) -> bool:
        return self.pos >= len(self.text)

    def skip_ws(self) -> None:
        while not self.eof() and self.peek() in _WS_CHARS:
            self.advance()

    def skip_to(self, c: str) -> bool:
        while not self.eof() and self.peek() != c:
            self.advance()
        if self.eof():
            return False
        self.advance()
        return True

    def expect(self, c: str) -> bool:
        if self.peek() == c:
            self.advance()
            return True
        return False

    def expect_str(self, s: str) -> bool:
        start = self.pos
        for ch in s:
            if self.peek() != ch:
                self.pos = start
                return False
            self.advance()
        return True


def _parse_value(it: _CharIterator) -> tuple[Any, int]:
    it.skip_ws()
    if it.eof():
        raise JSONParseError("unexpected end of input")
    c = it.peek()
    if c == "{":
        return _parse_object(it)
    if c == "[":
        return _parse_array(it)
    if c == '"':
        return _parse_string(it)
    if c == "t" or c == "f":
        return _parse_bool(it)
    if c == "n":
        return _parse_null(it)
    if c == "-" or c in _DIGIT_CHARS:
        return _parse_number(it)
    raise JSONParseError(f"unexpected character {c!r} at position {it.pos}")


def _parse_object(it: _CharIterator) -> tuple[dict, int]:
    if not it.expect("{"):
        raise JSONParseError(f"expected '{{' at {it.pos}")
    out: dict = {}
    it.skip_ws()
    if it.peek() == "}":
        it.advance()
        return out, it.pos
    while True:
        it.skip_ws()
        if it.peek() != '"':
            raise JSONParseError(f"expected string key at {it.pos}")
        key, _ = _parse_string(it)
        it.skip_ws()
        if not it.expect(":"):
            raise JSONParseError(f"expected ':' at {it.pos}")
        it.skip_ws()
        value, _ = _parse_value(it)
        # Use direct dict update to avoid rebuilding
        out[key] = value
        it.skip_ws()
        if it.eof():
            raise JSONParseError("unterminated object")
        c = it.peek()
        if c == ",":
            it.advance()
            continue
        if c == "}":
            it.advance()
            return out, it.pos
        raise JSONParseError(f"expected ',' or '}}' at {it.pos}")


def _parse_array(it: _CharIterator) -> tuple[list, int]:
    if not it.expect("["):
        raise JSONParseError(f"expected '[' at {it.pos}")
    out: list = []
    it.skip_ws()
    if it.peek() == "]":
        it.advance()
        return out, it.pos
    while True:
        it.skip_ws()
        value, _ = _parse_value(it)
        # Use direct list append to avoid rebuilding
        out.append(value)
        it.skip_ws()
        if it.eof():
            raise JSONParseError("unterminated array")
        c = it.peek()
        if c == ",":
            it.advance()
            continue
        if c == "]":
            it.advance()
            return out, it.pos
        raise JSONParseError(f"expected ',' or ']' at {it.pos}")


def _parse_string(it: _CharIterator) -> tuple[str, int]:
    if not it.expect('"'):
        raise JSONParseError(f"expected '\"' at {it.pos}")
    out = ""
    while not it.eof():
        c = it.advance()
        if c == '"':
            return out, it.pos
        if c == "\\":
            if it.eof():
                raise JSONParseError("unterminated string escape")
            esc = it.advance()
            if esc == '"':
                out = out + '"'
            elif esc == "\\":
                out = out + "\\"
            elif esc == "/":
                out = out + "/"
            elif esc == "b":
                out = out + "\b"
            elif esc == "f":
                out = out + "\f"
            elif esc == "n":
                out = out + "\n"
            elif esc == "r":
                out = out + "\r"
            elif esc == "t":
                out = out + "\t"
            elif esc == "u":
                if it.pos + 4 > len(it.text):
                    raise JSONParseError(f"bad unicode escape at {it.pos}")
                hex4 = it.text[it.pos : it.pos + 4]
                if not _is_hex4(hex4):
                    raise JSONParseError(f"bad unicode escape at {it.pos}")
                cp = int(hex4, 16)
                if 0xD800 <= cp <= 0xDBFF:
                    if it.pos + 5 >= len(it.text) or it.text[it.pos + 5] != "\\" or it.pos + 6 >= len(it.text) or it.text[it.pos + 6] != "u":
                        raise JSONParseError(f"unpaired high surrogate at {it.pos}")
                    if it.pos + 10 >= len(it.text):
                        raise JSONParseError(f"bad low surrogate at {it.pos}")
                    hex4b = it.text[it.pos + 7 : it.pos + 11]
                    if not _is_hex4(hex4b):
                        raise JSONParseError(f"bad low surrogate at {it.pos + 7}")
                    cp2 = int(hex4b, 16)
                    if not (0xDC00 <= cp2 <= 0xDFFF):
                        raise JSONParseError(f"bad low surrogate at {it.pos + 7}")
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (cp2 - 0xDC00)
                    out = out + chr(cp)
                    it.pos += 11
                    continue
                out = out + chr(cp)
                it.pos += 4
            else:
                raise JSONParseError(f"bad escape \\{esc} at {it.pos}")
            continue
        if ord(c) < 0x20:
            raise JSONParseError(f"unescaped control character at {it.pos}")
        out = out + c
    raise JSONParseError("unterminated string")

        return True, it.pos

        return False, it.pos

    start = it.pos
    if it.peek() == "-":
        it.advance()
    if it.eof():
        raise JSONParseError(f"bad number at {it.pos}")
    if it.peek() == "0":
        it.advance()
    elif it.peek() in _DIGIT_CHARS:
        while not it.eof() and it.peek() in _DIGIT_CHARS:
            it.advance()
    else:
        raise JSONParseError(f"bad number at {it.pos}")
